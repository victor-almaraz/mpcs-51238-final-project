1/// @file
2/// The floating-point mode of every thread that runs nodes (AGENTS.md,
3/// real-time rules; `threads.md` §6.6): round to nearest, exceptions
4/// masked, and flush-to-zero with denormals-are-zero. Decaying filter and
5/// delay tails reach denormal values, which are slow on most CPUs; the
6/// mode treats them as zero. Every such thread sets the whole control
7/// word, so none inherits a rounding mode or a trap from the thread that
8/// made it, and every thread computes the same bits.

23/// The control word of a thread that runs nodes: MXCSR with flush-to-zero
24/// (bit 15), every exception masked (bits 7 to 12), round to nearest
25/// (bits 13 and 14 clear), denormals-are-zero (bit 6), and no exception
26/// flag set.
27inline constexpr std::uint64_t nodeFloatControl = 0x9FC0;
28/// The bits of the control word that record exceptions that have
29/// happened (MXCSR bits 0 to 5), rather than set the mode: arithmetic sets
30/// them.
31inline constexpr std::uint64_t floatExceptionFlags = 0x3F;
32#elif defined(__aarch64__)
33/// The control word of a thread that runs nodes: FPCR with flush-to-zero
34/// (FZ, bit 24) alone, so round to nearest (RMode, bits 22 and 23, clear)
35/// and no exception trap enabled (bits 8 to 12 and 15 clear).
36inline constexpr std::uint64_t nodeFloatControl = std::uint64_t{1} << 24U;
37/// The bits of the control word that record exceptions: none, since
38/// FPSR, not FPCR, holds them.
39inline constexpr std::uint64_t floatExceptionFlags = 0;
40#else
41/// The control word of a thread that runs nodes: none on this processor.
42inline constexpr std::uint64_t nodeFloatControl = 0;
43/// The bits of the control word that record exceptions: none.
44inline constexpr std::uint64_t floatExceptionFlags = 0;
45#endif

47/// The calling thread's floating-point control word: MXCSR on x86, FPCR on
48/// AArch64, 0 elsewhere.
49inline std::uint64_t floatControl() noexcept ESPRESSO_NONBLOCKING {
50#if defined(__SSE__) || defined(__x86_64__) || defined(_M_X64)
51 return _mm_getcsr();
52#elif defined(__aarch64__)
53 std::uint64_t word = 0;
54 asm volatile("mrs %0, fpcr" : "=r"(word));
55 return word;
56#else
57 return 0;
58#endif
59}

61/// The calling thread's floating-point mode: its control word without the
62/// exception flags (floatExceptionFlags). A thread that runs nodes has
63/// mode nodeFloatControl.
64inline std::uint64_t floatMode() noexcept ESPRESSO_NONBLOCKING {
65 return floatControl() & ~floatExceptionFlags;
66}
67
68/// Sets the calling thread's floating-point control word to @p word (see
69/// floatControl()).
70inline void setFloatControl([[maybe_unused]] std::uint64_t word) noexcept ESPRESSO_NONBLOCKING {
71#if defined(__SSE__) || defined(__x86_64__) || defined(_M_X64)
72 _mm_setcsr(static_cast<unsigned>(word));
73#elif defined(__aarch64__)
74 asm volatile("msr fpcr, %0" : : "r"(word));
75#endif
76}
