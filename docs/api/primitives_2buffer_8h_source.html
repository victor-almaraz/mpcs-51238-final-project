26/// Decoded audio: what a sample loader returns (`samples.md` §3.3), at
27/// the rate it was asked for or, asked for none, at the file's own.
28struct decodedAudio {
29 /// One vector of samples per channel, all of one length.

34 /// The length, in frames: that of the first channel, or 0.
35 std::size_t frames() const noexcept { return channels.empty() ? 0 : channels.front().size(); }

116/// Channel @p to of @p toChannels, taken from a source of @p fromChannels
117/// channels whose channel c is @p get(c) (`samples.md` §4.4): with equal
118/// counts, channel @p to; from a mono source, its one channel; into a mono
119/// destination, the average; otherwise the same channel, or silence past
120/// the source's last.
121template <typename Get>
122float mapChannel(int to, int toChannels, int fromChannels, Get get) noexcept ESPRESSO_NONBLOCKING {
123 if (fromChannels == toChannels || fromChannels == 1) return get(fromChannels == 1 ? 0 : to);
124 if (toChannels == 1) {
125 float sum = 0.0f;
126 for (int c = 0; c < fromChannels; ++c) sum += get(c);
127 return sum / static_cast<float>(fromChannels);
128 }
129 return to < fromChannels ? get(to) : 0.0f;
130}

151 /// What makes an incoming buffer the same as a live one (§5.4).
152 struct key {

156 int channels = 1;
157 /// The declared length, in seconds, if any.

164 /// Ordered by every field.
165 friend bool operator<(const key& a, const key& b) {
