20/// What an event carries.
21enum class eventType : std::uint8_t {
22 noteOn = 1, ///< MIDI note on: channel, note, velocity
23 noteOff, ///< MIDI note off: channel, note, velocity
24 controlChange, ///< MIDI control change: channel, controller, value
25 pitchBend, ///< MIDI pitch bend: channel, 14-bit value
26 programChange, ///< MIDI program change: channel, program
27 channelPressure, ///< MIDI channel pressure: channel, pressure
28 polyPressure, ///< MIDI polyphonic pressure: channel, note, pressure
29 trigger, ///< A trigger with a value, usually 1
30 value, ///< A number
31 osc, ///< An OSC message stored in the pipe's byte pool
32};

34/// One timestamped event (`primitives.md` §5.4.1).
35struct event {
36 /// Position in the block, from 0 to blockSize − 1.

57/// True for the MIDI event types.
