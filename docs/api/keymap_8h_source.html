1/// @file
2/// Key binding schemes as data, and the state machine that resolves key
3/// sequences to command names (`tui.md` §8.1–8.5). Has no terminal
4/// dependency.
5///
6/// Keys use Emacs notation: `C-x` (Control), `M-f` (Meta), `S-<left>`
7/// (Shift), `C-M-_`, named keys such as `RET`, `TAB`, `DEL`
8/// (backspace), `ESC`, `SPC`, `<delete>`, `<up>`, `<home>`, `<prior>`
9/// (Page Up), `<next>` (Page Down), `<f5>`, and printable characters as
10/// themselves. A sequence is keys separated by spaces: `C-x C-s`.

79/// Resolves keys to commands for one scheme, tracking prefix keys and
80/// the vim mode.
81class keySequencer {
82public:
83 /// Outcome of one key.
84 enum class outcome : std::uint8_t {
85 command, ///< A binding matched; see result::command
86 pending, ///< The keys so far are a prefix of a binding
87 selfInsert, ///< Unbound printable key in a text-inserting mode
88 unbound, ///< No binding; the sequence is discarded
89 };
90
91 /// Outcome of one key, with the command or the keys typed.
92 struct result {
93 /// What happened.
94 outcome kind = outcome::unbound;
95 std::string command; ///< Command name for outcome::command
96 std::string keys; ///< The full sequence typed, such as "C-x C-s"
97 };
98
99 /// Creates a sequencer for @p scheme, in its initial mode.
100 explicit keySequencer(workspace::keyScheme scheme = workspace::keyScheme::emacs);
101
102 /// The key scheme.
103 workspace::keyScheme scheme() const noexcept { return scheme_; }
104 /// The current mode.
105 editMode mode() const noexcept { return mode_; }
106 /// Changes the mode; schemes other than vim stay in insert mode.
107 void setMode(editMode mode) noexcept;
108
109 /// True when unbound printable keys insert text in the current mode.
110 bool insertsText() const noexcept { return mode_ == editMode::insert; }
111
112 /// Keys typed so far in an incomplete sequence, such as "C-x".
113 std::string pendingKeys() const;
114
115 /// Processes one key.
116 result feed(std::string_view key);
117
118 /// Discards a pending sequence.
119 void reset() noexcept { pending_.clear(); }
120
121private:
122 workspace::keyScheme scheme_ = workspace::keyScheme::emacs;

125};
