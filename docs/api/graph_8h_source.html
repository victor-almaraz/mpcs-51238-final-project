90/// A parameter value: a number in canonical units, a choice or text, a
91/// list of numbers, or a bound pipe (`primitives.md` §7.1.2).
92using paramValue = std::variant<double, std::string, std::vector<double>, pipeRef>;
93
94/// The alternative @p v holds when it is a T: a checked view instead of
95/// `std::get_if`, which returns a pointer.
96template <typename T>
97optionalRef<T> as(const paramValue& v) noexcept {
98 if (!std::holds_alternative<T>(v)) return std::nullopt;
99 return std::cref(std::get<T>(v));
100}
101
102/// Deleted: the view would outlive the temporary it refers to.
103template <typename T>
104optionalRef<T> as(const paramValue&&) = delete;
