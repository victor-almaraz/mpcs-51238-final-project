25/// A node that cannot be built. Thrown on the builder thread only.
26class buildError : public std::runtime_error {
27public:
28 /// What an error is about.
29 enum class subject : std::uint8_t {
30 node, ///< A node, or the plan as a whole when the ID is empty
31 buffer, ///< A buffer declaration (`samples.md` §4.6)
32 };
33
34 /// An error in node @p node; empty for plan-level errors.
35 buildError(std::string node, const std::string& message)
36 : std::runtime_error(message), node_(std::move(node)) {}
37 /// An error about @p what with ID @p id.
38 buildError(subject what, std::string id, const std::string& message)
39 : std::runtime_error(message), node_(std::move(id)), about_(what) {}
40 /// The failing node's or buffer's ID.
41 const std::string& node() const noexcept { return node_; }
42 /// Whether node() names a node or a buffer.
43 subject about() const noexcept { return about_; }
44
45private:
46 std::string node_;
47 subject about_ = subject::node;
48};

61/// An executable graph: primitive instances in execution order, the
62/// pipes they use, and their I/O ports. Shared between the engine's
63/// threads; its last owner is always on the control thread.
64class plan {
65public:
66 /// Plan ID assigned by the client.
67 std::int32_t id = 0;
68 /// Primitives, in execution order (§10.1). Shared with the transitions
69 /// that run them during a fade.
70 std::vector<std::shared_ptr<primitive>> nodes;
71 /// Per-node data, parallel to `nodes`.
72 std::vector<nodeInfo> info;
73 /// Video-domain primitives, in execution order: the nodes that read or
74 /// write a video pipe. The render thread runs them once per frame; the
75 /// audio thread never does (§3).
76 std::vector<std::shared_ptr<primitive>> videoNodes;
77 /// The frame the visual layer shows: the input of the first
78 /// `videoOut`, or null.
79 std::shared_ptr<const vPipe> display;
80 /// The buffers the plan declares, in declaration order, held for as
81 /// long as the plan lives (`samples.md` §3.1).
82 std::vector<std::shared_ptr<sampleBuffer>> buffers;
83 /// Pipe declarations applied when the plan becomes active.
84 std::vector<metadata::pipeDecl> pipes;
85 /// Every pipe the plan touches, sorted and unique.
86 std::vector<metadata::pipeRef> used;
87 /// The plan's feedback pipes, sorted.
88 std::vector<metadata::pipeRef> feedback;
89 /// I/O ports of the plan's sources and sinks, shared with them.
90 std::vector<std::shared_ptr<ioPort>> ports;
91 /// Tempo to set on activation.
92 std::optional<double> tempo;
93 /// Beats per bar.
94 int meter = 4;
95 /// The page port of the graph it was built from (`io.md` §12.2): the
96 /// server listens on it while the plan is active.
97 std::optional<int> pagePort;
98
99 /// True when pipe @p ref is a feedback pipe in this plan.
100 bool isFeedback(metadata::pipeRef ref) const noexcept;
101};
