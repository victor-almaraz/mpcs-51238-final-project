19/// Owns all pipes. Allocates everything at construction (§6.2).
20class pipeTable {
21public:
22 /// Pipe counts per type and event pipe sizes.
23 struct capacity {
24 /// Audio pipes. Each holds `metadata::maxChannels` channels of a
25 /// block twice, 16 KB at 256 samples; the 64 layered voices of the
26 /// load piece use 915 (threads.md §9a.4).
27 int audio = 1024;
28 /// Control pipes, including the tempo pipe at index 0.
29 int control = 4096;
30 /// Event pipes.
31 int event = 128;
32 /// Events per event pipe per block.
33 int eventsPerPipe = 256;
34 /// Pool bytes per event pipe per block, for OSC messages.
35 int poolBytes = 8192;
36 /// Video pipes, each a frame of the engine's frame size. None by
37 /// default: a frame is megabytes, and only a server with a window
38 /// or an offline frame render needs them.
39 int video = 0;
40 };
41
42 /// Allocates every pipe for blocks of `info.blockSize`. The tempo
43 /// pipe starts at 120 BPM.
44 pipeTable(const capacity& limits, const prepareInfo& info);

76 /// Starts a block of @p numSamples for the audio and event pipes among
77 /// @p pipes (§10.2): zeroes the audio pipes and clears the event pipes.
78 /// Other kinds are skipped. Pipes of different calls may start on
79 /// different threads at once (`threads.md` §6.8).
80 void beginBlock(std::span<const metadata::pipeRef> pipes, int numSamples) noexcept ESPRESSO_NONBLOCKING;
81 /// Publishes the last samples of the audio pipes among @p pipes
82 /// (aPipe::publish). Other kinds are skipped.
83 void publish(std::span<const metadata::pipeRef> pipes) noexcept ESPRESSO_NONBLOCKING;
84 /// Ends a block for the audio and control pipes among @p pipes: each
85 /// audio pipe publishes its last samples, and feedback pipes swap
86 /// their buffers. Other kinds are skipped.
87 void endBlock(std::span<const metadata::pipeRef> pipes) noexcept ESPRESSO_NONBLOCKING;
88
89 /// Event writes dropped because a pipe or its pool was full.
90 std::uint64_t overflows() const noexcept { return overflows_->load(std::memory_order_relaxed); }
91
92private:
93 // Shared with every event pipe, so a pipe that outlives the table
94 // still counts into live memory.
95 ePipe::counter overflows_ = std::make_shared<std::atomic<std::uint64_t>>(0);
96 // Shared with the primitives bound to them, so a primitive's pipes
97 // live as long as it does, and a pipe never refers into the table.
98 std::vector<std::shared_ptr<aPipe>> audio_;
99 std::vector<std::shared_ptr<cPipe>> control_;
100 std::vector<std::shared_ptr<ePipe>> event_;
101 std::vector<std::shared_ptr<vPipe>> video_;
102};
