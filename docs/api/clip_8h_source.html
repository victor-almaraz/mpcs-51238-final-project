54/// Reads the Standard MIDI File @p bytes, format 0 or 1, into a clip: its
55/// tracks merged, a quarter note a beat by the file's division, its tempo
56/// and meter events ignored. A note-on of velocity 0 is a note-off; a note
57/// still held at the end of its track ends there. Nothing, with @p error
58/// set, for bytes that are not such a file, a file in SMPTE time, or more
59/// than maxClipNotes notes.

65/// The slots of one bank of clips, shared by the control thread, which
66/// gives it clips, and the node that plays it, which takes them at the
67/// start of a block (`canvas.md` §5.4). A clip a slot gives up goes back to
68/// the control thread, which frees it, so the node's thread never frees.
69class clipBank {
70public:
71 /// Clips given and not yet taken back, at most.
72 static constexpr std::size_t queueSize = 64;
73
74 clipBank() = default;
75 clipBank(const clipBank&) = delete;
76 clipBank& operator=(const clipBank&) = delete;
77 clipBank(clipBank&&) = delete;
78 clipBank& operator=(clipBank&&) = delete;
79 ~clipBank() = default;

81 /// Control thread: gives slot @p slot, from 0, the clip @p c, or none
82 /// when @p c is null. False, giving nothing, for a slot out of range or
83 /// while queueSize changes wait to be taken back (collect()).
84 [[nodiscard]] bool offer(int slot, std::unique_ptr<const clip> c);
85 /// Control thread: frees the clips the bank's node gave back.
86 void collect();
87
88 /// The thread that runs the bank's node: puts the clips offered since
89 /// the last call in their slots, giving back those they replace.
90 void update() noexcept ESPRESSO_NONBLOCKING;
91 /// The thread that runs the bank's node: slot @p slot's clip, if it
92 /// holds one.
93 metadata::optionalRef<clip> at(int slot) const noexcept ESPRESSO_NONBLOCKING;
94
95private:
96 struct change {
97 int slot = 0;
98 std::unique_ptr<const clip> item;
99 };
100 std::array<std::unique_ptr<const clip>, maxClipSlots> slots_;
101 // The change update() is taking, a member so that update() destroys
102 // none; empty between calls.
103 change taken_;
104 spscQueue<change> incoming_{queueSize};
105 // One item for each change taken, the clip it replaced or none, so it
106 // never fills: inFlight_ bounds the changes not yet given back.
107 spscQueue<std::unique_ptr<const clip>> outgoing_{queueSize};
108 std::size_t inFlight_ = 0; // control thread
109};
