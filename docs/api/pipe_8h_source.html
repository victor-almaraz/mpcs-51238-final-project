31/// Pipe interface (`primitives.md` §5.1). Primitives fetch a pipe's data
32/// once per block and never call virtual methods per sample.
33class pipe {
34public:
35 pipe() = default;
36 virtual ~pipe() = default;
37 // Pipes are shared through the pipe table, never copied or moved.
38 pipe(const pipe&) = delete;
39 pipe& operator=(const pipe&) = delete;
40 pipe(pipe&&) = delete;
41 pipe& operator=(pipe&&) = delete;
42
43 /// The pipe's type.
44 virtual metadata::pipeKind kind() const noexcept ESPRESSO_NONBLOCKING = 0;
45
46 /// Active channel count, from 1 to `metadata::maxChannels`.
47 int channels() const noexcept { return channels_; }
48 /// True when the pipe delays its data by one block (§5.6).
49 bool feedback() const noexcept { return feedback_; }
50
51 /// Sets the channel count, clamped to 1 to `metadata::maxChannels`, and
52 /// the feedback flag. Called on the audio thread when a plan becomes
53 /// active; storage is preallocated.
54 virtual void configure(int channels, bool feedback) noexcept ESPRESSO_NONBLOCKING;
55
56protected:
57 /// Active channel count.
58 int channels_ = 1;
59 /// Feedback flag.
60 bool feedback_ = false;
61};

63/// Audio pipe: `channels` × block samples. Writers sum; the engine zeroes
64/// the pipe at each block start (§5.2).
65class aPipe final : public pipe {
66public:
67 /// Preallocates @p capacity channels (1 to `maxChannels`) of
68 /// @p blockSize samples, twice when @p feedback is true (the second
69 /// buffer serves feedback). A pipe table's pipes take any shape a plan
70 /// declares; a plan's private pipe (`threads.md` §4.2) has its pipe's
71 /// channels and no feedback.
72 explicit aPipe(int blockSize, int capacity = metadata::maxChannels, bool feedback = true);
73
74 /// Audio.
75 metadata::pipeKind kind() const noexcept ESPRESSO_NONBLOCKING override {
76 return metadata::pipeKind::audio;
77 }
78 /// Sets the channel count, at most the capacity, and the feedback flag,
79 /// which a pipe without a second buffer ignores. A change of either
80 /// clears both buffers, so no channel or feedback buffer replays old
81 /// samples.
82 void configure(int channels, bool feedback) noexcept ESPRESSO_NONBLOCKING override;
83
84 /// Samples of @p channel in the current block; empty for a channel
85 /// outside the capacity. A feedback pipe returns the previous block's
86 /// data.
87 std::span<const float> read(int channel) const noexcept ESPRESSO_NONBLOCKING;
88
89 /// Adds @p src times @p gain into @p channel, or into the target's
90 /// @p channel while the pipe forwards (forward()). A channel outside
91 /// the active count is ignored.
92 void add(int channel, std::span<const float> src, float gain = 1.0f) noexcept ESPRESSO_NONBLOCKING;
93 /// Adds @p src into @p channel with a gain ramp from @p gainStart to
94 /// @p gainEnd across the block (crossfades, §9.3), or into the target's
95 /// @p channel while the pipe forwards (forward()). A channel outside the
96 /// active count is ignored.
97 void addRamp(int channel, std::span<const float> src, float gainStart,
98 float gainEnd) noexcept ESPRESSO_NONBLOCKING;
99
100 /// Lets a private pipe (`threads.md` §4.2) forward to @p target, the
101 /// table pipe its summing node adds it into, which outlives the plan
102 /// that owns this pipe. The builder calls it for the private pipes of
103 /// a group whose writers each add once a block (§4.2.5).
104 void forwardTo(aPipe& target) noexcept { target_ = target; }
105 /// Turns forwarding on or off; a pipe without a target never forwards.
106 /// While it forwards, add() and addRamp() add into the target, the
107 /// engine does not clear this pipe at a block's start, and its summing
108 /// node skips it. The target then holds the bits the summing node
109 /// would have left: a writer that adds w once into this pipe, cleared
110 /// to +0, leaves 0 + w, which the summing node adds with gain 1 into a
111 /// target that never holds −0, and p + (0 + w) equals p + w. That holds
112 /// only while each writer adds once a block, nothing reads the target
113 /// between the writers and the sum, and the writers and the sum run in
114 /// plan order: one plan alone, with no crossfade and no shuffled order
115 /// (`threads.md` §4.2.5). The engine turns it on and off for a whole
116 /// plan, on the audio thread between blocks.
117 void forward(bool on) noexcept ESPRESSO_NONBLOCKING { forwarding_ = on && target_.has_value(); }
118 /// True while add() and addRamp() go to the target.
119 bool forwarding() const noexcept { return forwarding_; }
120 /// True when the pipe has a target (forwardTo()), so that it forwards
121 /// whenever the engine turns forwarding on for its plan.
122 bool hasTarget() const noexcept { return target_.has_value(); }
123
124 /// Starts a block of @p numSamples: zeroes the buffer being written.
125 void beginBlock(int numSamples) noexcept ESPRESSO_NONBLOCKING;
126 /// Publishes each channel's last sample written in the block, which
127 /// latest() returns: once no writer of the block is left to run.
128 void publish() noexcept ESPRESSO_NONBLOCKING;
129 /// Ends a block: publishes each channel's last sample (publish()), and
130 /// a feedback pipe swaps its buffers.
131 void endBlock() noexcept ESPRESSO_NONBLOCKING;
132
133 /// Any thread: the last sample of @p channel at the end of the latest
134 /// block. The render thread reads audio through it (the window of one
135 /// sample of `primitives.md` §8.1.2's latestWindow mode).
136 float latest(int channel) const noexcept {
137 return latest_[static_cast<std::size_t>(std::clamp(channel, 0, metadata::maxChannels - 1))].load(
138 std::memory_order_relaxed);
139 }
140
141private:
142 // A feedback pipe reads one buffer while writers fill the other and
143 // swaps them at each block end; any other pipe uses the first only.
144 std::span<const float> readBuffer() const noexcept { return buffer(feedback_ && swapped_); }
145 std::span<float> writeBuffer() noexcept { return buffer(feedback_ && !swapped_); }
146 // The first buffer, or the second when @p second is true (only a
147 // pipe with two has a second).
148 std::span<float> buffer(bool second) const noexcept {
149 const auto size = samples_.size() / static_cast<std::size_t>(buffers_);
150 return samples_.subspan(second ? size : 0, size);
151 }
152
153 // The samples of @p channel in @p buffer.
154 template <typename Sample>
155 std::span<Sample> channelOf(std::span<Sample> buffer, int channel) const noexcept {
156 const auto size = static_cast<std::size_t>(blockSize_);
157 return buffer.subspan(static_cast<std::size_t>(channel) * size, size);
158 }
159 // The part of @p channel this block writes, or empty when the channel
160 // is not active; while the pipe forwards, the target's.
161 std::span<float> written(int channel) noexcept ESPRESSO_NONBLOCKING;
162
163 // First, beside the channel count of pipe: a write to a pipe that
164 // forwards reads those three from this pipe and nothing else, so it
165 // takes one cache line of it rather than two. A plan's many private
166 // pipes are otherwise untouched while they forward, so that line is
167 // rarely cached. Only a pipe that forwards reads target_, its group's
168 // table pipe, set when the group may forward.
169 bool forwarding_ = false;
170 std::optional<std::reference_wrapper<aPipe>> target_;
171 int blockSize_ = 0;
172 int numSamples_ = 0;
173 int capacity_ = metadata::maxChannels;
174 int buffers_ = 2;
175 // Every buffer, `capacity_` channels each, zeroed by one memset: a
176 // vector's element-wise construction made a default pipe table take
177 // 85 ms instead of 5 ms in Debug builds (modern-cpp skill §9).
178 std::unique_ptr<float[]> owner_; // NOLINT(modernize-avoid-c-arrays): skill §9
179 std::span<float> samples_;
180 bool swapped_ = false;
181 static_assert(realtimeAtomic<float>);
182 std::array<std::atomic<float>, metadata::maxChannels> latest_{};
183};

185/// Control pipe: one value per channel. A write replaces the value, which
186/// holds until the next write (§5.3). Values are atomics so other threads
187/// can read them (`/c.get`).
188class cPipe final : public pipe {
189public:
190 /// Control.
191 metadata::pipeKind kind() const noexcept ESPRESSO_NONBLOCKING override {
192 return metadata::pipeKind::control;
193 }
194 /// Sets the channel count and feedback flag. Channels that become
195 /// active take channel 0's value, which a reader of the narrower pipe
196 /// saw on every channel; a feedback change carries the current values
197 /// into both halves, so the next block end publishes no stale value.
198 void configure(int channels, bool feedback) noexcept ESPRESSO_NONBLOCKING override;
199
200 /// The value of @p channel. A feedback pipe returns the value written
201 /// in the previous block.
202 float value(int channel = 0) const noexcept {
203 return front_[clampChannel(channel)].load(std::memory_order_relaxed);
204 }
205
206 /// Sets @p channel to @p v.
207 void set(int channel, float v) noexcept ESPRESSO_NONBLOCKING;
208 /// Sets every channel to @p v.
209 void setAll(float v) noexcept ESPRESSO_NONBLOCKING;
210
211 /// Ends a block: a feedback pipe publishes the values written in it.
212 void endBlock() noexcept ESPRESSO_NONBLOCKING;
213
214private:
215 static std::size_t clampChannel(int c) noexcept {
216 return static_cast<std::size_t>(c < 0 ? 0
217 : (c >= metadata::maxChannels ? metadata::maxChannels - 1 : c));
218 }
219
220 // Value-initialized: every value starts at 0.
221 static_assert(realtimeAtomic<float>);
222 std::array<std::atomic<float>, metadata::maxChannels> front_{};
223 std::array<std::atomic<float>, metadata::maxChannels> back_{};
224};

240 return metadata::pipeKind::event;
241 }
242

244 /// writer order (§5.4.2).

248 /// overflow, and returns false (§5.4.3).
249 bool write(const event& e) noexcept ESPRESSO_NONBLOCKING;
250
251 /// Copies an OSC message into the pool and writes an OSC event for
252 /// it. Drops and counts it when the pool or the array is full.
253 bool writeOsc(std::uint32_t offset, std::span<const std::byte> message) noexcept ESPRESSO_NONBLOCKING;
254
255 /// The pool bytes of @p ref.
256 std::span<const std::byte> data(poolRef ref) const noexcept ESPRESSO_NONBLOCKING;
257
258 /// Copies @p e, and its pool data for an OSC event, from @p from.
259 bool copyFrom(const ePipe& from, const event& e) noexcept ESPRESSO_NONBLOCKING;
260
261 /// Starts a block: clears the events and the pool.
262 void beginBlock() noexcept ESPRESSO_NONBLOCKING;
263
264private:
265 std::optional<poolRef> store(std::span<const std::byte> bytes) noexcept ESPRESSO_NONBLOCKING;
266 void overflow() noexcept ESPRESSO_NONBLOCKING;
267
268 // Value-initialized arrays, not vectors, whose element-wise construction
269 // took 25 ms a table in Debug (modern-cpp skill §9). Construction writes
270 // every page, so the audio thread's first writes take no page fault.
271 std::unique_ptr<event[]> eventStore_; // NOLINT(modernize-avoid-c-arrays): skill §9
272 std::unique_ptr<std::byte[]> poolStore_; // NOLINT(modernize-avoid-c-arrays): skill §9
273 std::span<event> events_;
274 std::size_t count_ = 0;
275 std::span<std::byte> pool_;
276 std::size_t poolUsed_ = 0;
277 counter overflows_;
278};
