101/// The nodes of an execution order grouped into tasks (`threads.md` §5):
102/// a thread runs a task's nodes in order, start to finish, once every
103/// task it waits for has finished.
104struct taskList {
105 /// The nodes of every task, by position in the order: task by task,
106 /// each task's in execution order.
107 std::vector<std::uint32_t> nodes;
108 /// Where each task's nodes start in `nodes`; one entry more than there
109 /// are tasks.
110 std::vector<std::uint32_t> start{0};
111 /// The tasks that wait for each task, task by task, each task's sorted.
112 std::vector<std::uint32_t> successors;
113 /// Where each task's successors start in `successors`; one entry more
114 /// than there are tasks.
115 std::vector<std::uint32_t> successorStart{0};
116 /// The number of tasks each task waits for.
117 std::vector<std::uint32_t> waits;
118
119 /// The number of tasks.
120 std::size_t size() const noexcept { return start.size() - 1; }
121 /// The nodes of task @p t, which is less than size().
122 std::span<const std::uint32_t> nodesOf(std::size_t t) const noexcept {
123 return std::span(nodes).subspan(start[t], start[t + 1] - start[t]);
124 }
125 /// The tasks that wait for task @p t.
126 std::span<const std::uint32_t> successorsOf(std::size_t t) const noexcept {
127 return std::span(successors).subspan(successorStart[t], successorStart[t + 1] - successorStart[t]);
128 }
129};

141/// For each node of an execution order, the earlier nodes it must run
142/// after (`threads.md` §4.1). Every node a node must follow is reachable
143/// through the list, though not always directly: a node after two
144/// writers of a pipe lists only the second, which follows the first.
145struct dependencyList {
146 /// Where each node's dependencies start in `before`; one entry more
147 /// than there are nodes.
148 std::vector<std::uint32_t> start{0};
149
150 /// The dependencies of every node, by position in the order, each
151 /// node's sorted.
152 std::vector<std::uint32_t> before;
153
154 /// The number of nodes.
155 std::size_t size() const noexcept { return start.size() - 1; }
156 /// The dependencies of node @p i, which is less than size().
157 std::span<const std::uint32_t> of(std::size_t i) const noexcept {
158 return std::span(before).subspan(start[i], start[i + 1] - start[i]);
159 }
160};

162/// Items for each node of an execution order, node by node in one array.
163template <typename T>
164struct perNode {
165 /// Where each node's items start in `items`; one entry more than there
166 /// are nodes.
167 std::vector<std::uint32_t> start{0};
168 /// The items of every node, node by node.
169 std::vector<T> items;

172 std::size_t size() const noexcept { return start.size() - 1; }
173 /// The items of node @p i, which is less than size().
174 std::span<const T> of(std::size_t i) const noexcept {
175 return std::span(items).subspan(start[i], start[i + 1] - start[i]);
176 }

179/// The work a block does on its pipes (`threads.md` §6.8), placed so that
180/// most of it runs inside the tasks: each audio and event pipe of the
181/// table, and each private pipe, starts (zeroed or emptied for the block)
182/// just before the node that first uses it in the order runs, when that
183/// node writes it, and each audio pipe publishes its last samples just
184/// after the node that last writes it. The audio thread does the rest,
185/// before and after every node. Built on the builder thread with the
186/// order's schedule; the audio thread and the workers only read it.
187///
188/// A start in a node is race-free: the node is the first in the order to
189/// use the pipe, and every later user follows it through the dependencies
190/// (`threads.md` §4.1.2), in every order the dependencies admit. Feedback
191/// pipes, whose reads follow no writer, start and end on the audio thread.
192struct pipeWork {
193 /// The table's audio and event pipes each node starts before it runs.
194 perNode<metadata::pipeRef> starts;
195 /// The private pipes each node starts before it runs.
196 perNode<std::shared_ptr<aPipe>> privateStarts;
197 /// The table's audio pipes each node publishes after it runs
198 /// (aPipe::publish).
199 perNode<metadata::pipeRef> publishes;
200 /// The table's audio and event pipes the audio thread starts before any
201 /// node runs: feedback pipes, pipes no node writes, and pipes a node
202 /// reads before the first writes them.
203 std::vector<metadata::pipeRef> serialStarts;
204 /// The private pipes the audio thread starts before any node runs:
205 /// those no node of the order writes (in a crossfade, those of writers
206 /// the incoming plan adopted) and those a node reads before their
207 /// writer runs.
208 std::vector<std::shared_ptr<aPipe>> serialPrivateStarts;
209 /// The pipes the audio thread ends after every node has run
210 /// (pipeTable::endBlock): feedback audio and control pipes, and audio
211 /// pipes no node writes. The end of a control pipe that is not
212 /// feedback does nothing.
213 std::vector<metadata::pipeRef> serialEnds;
214};

216/// How the nodes of an execution order may run in parallel: their
217/// dependencies, their tasks, and the scheduler's counters for the tasks.
218struct nodeSchedule {
219 /// The dependencies of the nodes, in their order (`threads.md` §4.1).
220 dependencyList dependencies;
221 /// The nodes grouped into tasks (`threads.md` §5).
222 taskList tasks;
223 /// The scheduler's counters for `tasks`.
224 taskCounters counters;
225 /// True when some two tasks can run at once: some level of the task
226 /// graph holds two tasks. A schedule that is not parallel is a chain
227 /// of tasks, which the engine runs on the audio thread alone, in
228 /// `serialOrder`, waking no worker (`threads.md` §6.1).
229 bool parallel = false;
230 /// The order in which a thread that runs every node of a block alone
231 /// runs them (an engine of one thread, or a schedule that is not
232 /// parallel or does not fit the pool): empty for the order the nodes
233 /// are in, or, with `scheduleOptions::shuffleSeed`, a permutation of
234 /// their positions that keeps every dependency (`threads.md` §5.5).
235 std::vector<std::uint32_t> serialOrder;
236 /// The block's work on pipes, by node (`threads.md` §6.8). Plans and
237 /// crossfades fill it; a schedule made by itself leaves it empty.
238 pipeWork pipes;
239};
