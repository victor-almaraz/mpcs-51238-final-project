18/// A ring buffer with one producer thread and one consumer thread.
19/// Slots are preallocated; items move in and out, so a queue of
20/// `std::unique_ptr` hands ownership between threads without allocating.
21template <typename T>
22// head_ and tail_ each own a cache line, so the producer and consumer do
23// not share one; the analyzer's tighter order would put them together.
24// NOLINTNEXTLINE(clang-analyzer-optin.performance.Padding): cache-line separation
25class spscQueue {
26public:
27 /// Allocates room for at least @p capacity items.
28 explicit spscQueue(std::size_t capacity)
29 : slots_(std::bit_ceil(capacity < 2 ? std::size_t{2} : capacity)), mask_(slots_.size() - 1) {}
30
31 ~spscQueue() = default;
32 spscQueue(const spscQueue&) = delete;
33 spscQueue& operator=(const spscQueue&) = delete;
34 spscQueue(spscQueue&&) = delete;
35 spscQueue& operator=(spscQueue&&) = delete;
36
37 /// Producer: appends a copy of @p item. Returns false when the queue
38 /// is full.
39 [[nodiscard]] bool
40 push(const T& item) noexcept(std::is_nothrow_copy_assignable_v<T>) ESPRESSO_NONBLOCKING {
41 return put(item);
42 }
43
44 /// Producer: moves @p item in. Returns false when the queue is full, in
45 /// which case @p item is untouched.
46 [[nodiscard]] bool push(T&& item) noexcept ESPRESSO_NONBLOCKING { return put(std::move(item)); }
47
48 /// Consumer: moves the oldest item into @p out, which holds nothing to
49 /// free (an empty or moved-from owner). Returns false when the queue
50 /// is empty.
51 [[nodiscard]] bool pop(T& out) noexcept ESPRESSO_NONBLOCKING {
52 const std::size_t head = head_.load(std::memory_order_relaxed);
53 if (head == tail_.load(std::memory_order_acquire)) return false;
54 ESPRESSO_ASSUME_NONBLOCKING_BEGIN // out holds nothing to free (above)
55 out = std::move(slots_[head & mask_]);
56 ESPRESSO_ASSUME_NONBLOCKING_END
57 head_.store(head + 1, std::memory_order_release);
58 return true;
59 }
60
61 /// Consumer: true when there is nothing to pop.
62 bool empty() const noexcept ESPRESSO_NONBLOCKING {
63 return head_.load(std::memory_order_relaxed) == tail_.load(std::memory_order_acquire);
64 }
65
66 /// Number of slots.
67 std::size_t capacity() const noexcept ESPRESSO_NONBLOCKING { return slots_.size(); }
68
69private:
70 // Assigns the slot directly from @p item: one copy or one move.
71 template <typename U>
72 bool put(U&& item) ESPRESSO_NONBLOCKING {
73 const std::size_t tail = tail_.load(std::memory_order_relaxed);
74 if (tail - head_.load(std::memory_order_acquire) == slots_.size()) return false;
75 // The slot holds nothing to free: empty since construction, or moved
76 // from by the pop that last took its item.
77 ESPRESSO_ASSUME_NONBLOCKING_BEGIN
78 slots_[tail & mask_] = std::forward<U>(item);
79 ESPRESSO_ASSUME_NONBLOCKING_END
80 tail_.store(tail + 1, std::memory_order_release);
81 return true;
82 }
83
84 std::vector<T> slots_;
85 const std::size_t mask_;
86 alignas(64) std::atomic<std::size_t> head_{0};
87 alignas(64) std::atomic<std::size_t> tail_{0};
88};
