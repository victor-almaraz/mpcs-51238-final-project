31/// Server state shown in the status bar (`tui.md` §3, §4).
32struct serverStatus {
33 /// True after a successful `/hello`.
34 bool connected = false;
35 std::string server; ///< Server name and version from `/hello.ok`
36 /// Tempo of the server's beat clock, in beats per minute, from
37 /// `/counters`; empty until the server reports one.
38 std::optional<double> bpm;
39 /// Position of the server's beat clock, in beats, from `/counters`;
40 /// empty until the server reports one.
41 std::optional<double> beat;
42 /// Beats per bar of the active plan, from `/counters`.
43 std::int32_t meter = 4;
44 /// Missed deadlines.
45 std::int32_t missed = 0;
46 /// Device underruns and overruns.
47 std::int32_t xruns = 0;
48 /// Dropped requests.
49 std::int32_t drops = 0;
50 /// Ring buffer overflows.
51 std::int32_t overflows = 0;
52 /// Audio thread load, from 0 to 1.
53 float load = 0.0f;
54 /// The threads the server runs a block on, from `/counters`
55 /// (`threads.md` §8).
56 std::int32_t threads = 1;
57 /// Each thread's load, the audio thread first; empty from a server that
58 /// does not report them.
59 std::vector<float> threadLoads;
60 /// The active plan, or 0.
61 std::int32_t activePlan = 0;
62
63 /// The busiest thread's load: the largest of `threadLoads`, or `load`
64 /// when there are none.
65 float busiest() const;
66 /// The load as the status bar shows it: `Load 45%`, the busiest
67 /// thread's, then ` · Threads 4` for more than one thread.
68 std::string loadText() const;
69 /// The tempo as the status bar shows it: `120 BPM`, to one decimal
70 /// place, or `— BPM` when not connected or not reported.
71 std::string tempoText() const;
72 /// The beat position as the status bar shows it: `bar.beat`, both
73 /// counted from 1, such as `9.3`; or `—` when not connected, when no
74 /// plan is active, or when not reported.
75 std::string beatText() const;
76};
