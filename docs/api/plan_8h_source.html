38 /// An error in node @p node; empty for plan-level errors.
39 buildError(std::string node, const std::string& message)

99/// The nodes of an execution order grouped into tasks (`threads.md` §5):
100/// a thread runs a task's nodes in order, start to finish, once every
101/// task it waits for has finished.
102struct taskList {
103 /// The nodes of every task, by position in the order: task by task,
104 /// each task's in execution order.
105 std::vector<std::uint32_t> nodes;
106 /// Where each task's nodes start in `nodes`; one entry more than there
107 /// are tasks.
108 std::vector<std::uint32_t> start{0};
109 /// The tasks that wait for each task, task by task, each task's sorted.
110 std::vector<std::uint32_t> successors;
111 /// Where each task's successors start in `successors`; one entry more
112 /// than there are tasks.
113 std::vector<std::uint32_t> successorStart{0};
114 /// The number of tasks each task waits for.
115 std::vector<std::uint32_t> waits;
116
117 /// The number of tasks.
118 std::size_t size() const noexcept { return start.size() - 1; }
119 /// The nodes of task @p t, which is less than size().
120 std::span<const std::uint32_t> nodesOf(std::size_t t) const noexcept {
121 return std::span(nodes).subspan(start[t], start[t + 1] - start[t]);
122 }
123 /// The tasks that wait for task @p t.
124 std::span<const std::uint32_t> successorsOf(std::size_t t) const noexcept {
125 return std::span(successors).subspan(successorStart[t], successorStart[t + 1] - successorStart[t]);
126 }
127};

129/// What the scheduler counts down in each block (`threads.md` §6.1): for

139/// For each node of an execution order, the earlier nodes it must run
140/// after (`threads.md` §4.1). Every node a node must follow is reachable
141/// through the list, though not always directly: a node after two
142/// writers of a pipe lists only the second, which follows the first.
143struct dependencyList {
144 /// Where each node's dependencies start in `before`; one entry more
145 /// than there are nodes.
146 std::vector<std::uint32_t> start{0};
147
148 /// The dependencies of every node, by position in the order, each
149 /// node's sorted.
150 std::vector<std::uint32_t> before;
151
152 /// The number of nodes.
153 std::size_t size() const noexcept { return start.size() - 1; }
154 /// The dependencies of node @p i, which is less than size().
155 std::span<const std::uint32_t> of(std::size_t i) const noexcept {
156 return std::span(before).subspan(start[i], start[i + 1] - start[i]);
157 }
158};

160/// How the nodes of an execution order may run in parallel: their
161/// dependencies, their tasks, and the scheduler's counters for the tasks.
162struct nodeSchedule {
163 /// The dependencies of the nodes, in their order (`threads.md` §4.1).
164 dependencyList dependencies;
165 /// The nodes grouped into tasks (`threads.md` §5).
166 taskList tasks;
167 /// The scheduler's counters for `tasks`.
168 taskCounters counters;
169 /// True when some two tasks can run at once: some level of the task
170 /// graph holds two tasks. A schedule that is not parallel is a chain
171 /// of tasks, which the engine runs on the audio thread alone, in
172 /// `serialOrder`, waking no worker (`threads.md` §6.1).
173 bool parallel = false;
174 /// The order in which a thread that runs every node of a block alone
175 /// runs them (an engine of one thread, or a schedule that is not
176 /// parallel or does not fit the pool): empty for the order the nodes
177 /// are in, or, with `scheduleOptions::shuffleSeed`, a permutation of
178 /// their positions that keeps every dependency (`threads.md` §5.5).
179 std::vector<std::uint32_t> serialOrder;
180};

182/// The least estimated cost of a task by default, in nanoseconds a block

187/// How the builder makes a schedule (`threads.md` §5.2, §5.5), from

191 double minimumTaskCost = defaultMinimumTaskCost;
192 /// When set, the seed of `nodeSchedule::serialOrder`.

199class plan {
200public:

203 /// Primitives, in execution order (§10.1). Shared with the transitions

205 std::vector<std::shared_ptr<primitive>> nodes;

207 std::vector<nodeInfo> info;

209 /// cleared at each block start with the pipe table's.
210 std::vector<std::shared_ptr<aPipe>> privatePipes;
211 /// How `nodes` may run in parallel (`threads.md` §4, §5).
212 nodeSchedule schedule;
213 /// Seconds each node has spent in process(), parallel to `nodes`, when
214 /// the engine profiles (`engineOptions::profile`); written by the thread
215 /// that runs the node, the audio thread or a worker, and read once the
216 /// plan has stopped (`threads.md` §11).
217 std::vector<double> seconds;
218 /// Video-domain primitives, in execution order: the nodes that read or
219 /// write a video pipe. The render thread runs them once per frame; the
220 /// audio thread and the workers never do (§3).
221 std::vector<std::shared_ptr<primitive>> videoNodes;
222 /// The frame the visual layer shows: the input of the first
223 /// `videoOut`, or null.
224 std::shared_ptr<const vPipe> display;
225 /// The buffers the plan declares, in declaration order, held for as
226 /// long as the plan lives (`samples.md` §3.1).
227 std::vector<std::shared_ptr<sampleBuffer>> buffers;
228 /// Pipe declarations applied when the plan becomes active.
229 std::vector<metadata::pipeDecl> pipes;
230 /// Every pipe the plan touches, sorted and unique.
231 std::vector<metadata::pipeRef> used;
232 /// The plan's feedback pipes, sorted.
233 std::vector<metadata::pipeRef> feedback;
234 /// I/O ports of the plan's sources and sinks, shared with them.
235 std::vector<std::shared_ptr<ioPort>> ports;
236 /// Tempo to set on activation.
237 std::optional<double> tempo;
238 /// Beats per bar.
239 int meter = 4;
240 /// The page port of the graph it was built from (`io.md` §12.2): the
241 /// server listens on it while the plan is active.
242 std::optional<int> pagePort;
243
244 /// True when pipe @p ref is a feedback pipe in this plan.
245 bool isFeedback(metadata::pipeRef ref) const noexcept;
246};
