48/// What a primitive sees during one block (§4.4).
49struct blockContext {
50 /// Engine sample clock at the block start.
51 std::int64_t blockStartSample = 0;
52 /// Samples in this block, at most `prepareInfo::blockSize`.
53 int numSamples = 0;
54 /// Sample rate, in hertz.
55 double sampleRate = 48000.0;
56 /// Beat position at the block start (§7.5).
57 double beat = 0.0;
58 /// Tempo read at the block start, in beats per minute.
59 double tempo = 120.0;
60 /// Beats per bar.
61 int meter = 4;
62 /// Device clock at the block start, in seconds. Sources convert
63 /// receive timestamps to sample offsets against it (§7.2.2).
64 double hostTime = 0.0;
65 /// Device buffers.
66 deviceBuffers device;
67 /// Index of this block's first frame within the device buffers.
68 int deviceOffset = 0;
69
70 /// Beats advanced per sample at this block's tempo.
71 double beatsPerSample() const noexcept { return tempo / 60.0 / sampleRate; }
72};

74/// Base of every source, sink, and generator (§4.1). The builder
75/// constructs and prepares a primitive; the audio thread calls
76/// adoptState() at a swap; the thread that runs the node (the audio
77/// thread or a worker, `threads.md` §2) calls process() once per block,
78/// after the nodes it depends on and before the block ends; the message
79/// thread destroys it.
80class primitive {
81public:
82 /// A primitive with node ID @p id of type @p type. @p type must
83 /// outlive the primitive (it names a `metadata` table entry).
84 primitive(std::string id, std::string_view type) : id_(std::move(id)), type_(type) {}
85 virtual ~primitive() = default;
86
87 primitive(const primitive&) = delete;
88 primitive& operator=(const primitive&) = delete;
89
90 /// Allocates every buffer process() uses (builder thread). The
91 /// built-in primitives allocate when constructed, on the same thread,
92 /// so the default does nothing.
93 virtual void prepare(const prepareInfo& info) { (void)info; }
94
95 /// Takes over the state of @p outgoing, a primitive of the same type
96 /// and shape from the outgoing plan (§9.2). Runs on the audio thread
97 /// at the swap, so it swaps or copies preallocated buffers only.
98 virtual void adoptState(primitive& outgoing) noexcept ESPRESSO_NONBLOCKING { (void)outgoing; }
99
100 /// Processes one block (§4.4).
101 virtual void process(const blockContext& ctx) noexcept ESPRESSO_NONBLOCKING = 0;
102
103 /// Node ID.
104 const std::string& id() const noexcept { return id_; }
105 /// Primitive type.
106 std::string_view type() const noexcept { return type_; }
107
108 /// Sets the fade for the next block: output gain ramps from
109 /// @p gainStart to @p gainEnd. A retiring primitive's control writes
110 /// and its events other than note-offs are discarded (§9.3.4).
111 /// @p finalBlock marks a retiring primitive's last block. Called by
112 /// the engine.
113 void setFade(float gainStart, float gainEnd, bool retiring,
114 bool finalBlock = false) noexcept ESPRESSO_NONBLOCKING {
115 gainStart_ = gainStart;
116 gainEnd_ = gainEnd;
117 retiring_ = retiring;
118 finalBlock_ = finalBlock;
119 }
120
121 /// True while the primitive is fading out.
122 bool retiring() const noexcept { return retiring_; }
123 /// True in a retiring primitive's last block: the time to end held notes.
124 bool finalBlock() const noexcept { return finalBlock_; }
125 /// True when @p e may be written: always, or only note-offs while
126 /// retiring, so notes held across a swap still end.
127 bool passes(const event& e) const noexcept ESPRESSO_NONBLOCKING;
128
129protected:
130 /// Adds @p samples to @p channel of @p out with the fade applied.
131 void emit(aPipe& out, int channel, std::span<const float> samples) noexcept ESPRESSO_NONBLOCKING;
132 /// Sets @p channel of @p out, unless retiring.
133 void emit(cPipe& out, int channel, float value) noexcept ESPRESSO_NONBLOCKING;
134 /// Writes @p e to @p out, unless retiring; a retiring primitive still
135 /// writes note-offs (passes()).
136 bool emit(ePipe& out, const event& e) noexcept ESPRESSO_NONBLOCKING;
137 /// Copies @p e, with its OSC data, from @p from to @p out when it
138 /// passes().
139 bool forward(ePipe& out, const ePipe& from, const event& e) noexcept ESPRESSO_NONBLOCKING;
140 /// Writes the OSC message @p message to @p out at @p offset, unless
141 /// retiring.
142 bool emitOsc(ePipe& out, std::uint32_t offset,
143 std::span<const std::byte> message) noexcept ESPRESSO_NONBLOCKING;
144 /// Adds @p samples to a device buffer with the fade applied.
145 void emitDevice(std::span<float> out, std::span<const float> samples) noexcept ESPRESSO_NONBLOCKING;
146
147private:
148 std::string id_;
149 std::string_view type_;
150 float gainStart_ = 1.0f;
151 float gainEnd_ = 1.0f;
152 bool retiring_ = false;
153 bool finalBlock_ = false;
154};
