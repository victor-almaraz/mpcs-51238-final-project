24/// Severity or origin of a console entry: `info` for the editor's and
25/// interpreter's messages, `error` for errors, `server` for server messages.

28/// One console line.
29struct consoleEntry {
30 /// Severity or origin.
31 consoleKind kind = consoleKind::info;
32 /// The line of text.
33 std::string text;
34 /// Where clicking the entry moves the cursor, if anywhere.

36};

38/// Bounded list of console entries, oldest first.
39class consoleLog {
40public:
41 /// Creates a log that keeps the newest @p capacity entries.
42 explicit consoleLog(std::size_t capacity = 1000) : capacity_(capacity) {}
43
44 /// Appends an entry, dropping the oldest beyond the capacity.

46 /// Appends an informational entry.
47 void info(std::string text) { add(consoleKind::info, std::move(text)); }
48 /// Appends an error, optionally pointing to its location.

50 add(consoleKind::error, std::move(text), std::move(location));
51 }
52 /// Removes all entries.
