19/// A position in a buffer. Both fields are 0-based; the column counts
20/// code points.
21struct position {
22 /// Line, from 0.
23 std::size_t line = 0;
24 /// Column in code points, from 0.
25 std::size_t column = 0;
26
27 /// Equal when line and column are equal.
28 friend bool operator==(const position&, const position&) = default;
29 /// Ordered by line, then column.
30 friend auto operator<=>(const position&, const position&) = default;
31};

33/// Byte storage with a movable gap, so that edits at the cursor are
34/// cheap (`tui.md` §5.1).
35class gapBuffer {
36public:
37 gapBuffer() = default;
38 /// Creates a buffer holding @p text.
39 explicit gapBuffer(std::string_view text);
40
41 /// Number of bytes stored.
42 std::size_t size() const noexcept { return data_.size() - gapLength(); }
43
44 /// Byte at @p offset. Requires offset < size().
45 char at(std::size_t offset) const noexcept;
46
47 /// Copy of @p length bytes starting at @p offset, clamped to size().
48 std::string substr(std::size_t offset, std::size_t length) const;
49
50 /// Copy of the whole contents.
51 std::string str() const { return substr(0, size()); }
52
53 /// Inserts @p text at @p offset.
54 void insert(std::size_t offset, std::string_view text);
55
56 /// Removes @p length bytes starting at @p offset.
57 void erase(std::size_t offset, std::size_t length);
58
59private:
60 std::size_t gapLength() const noexcept { return gapEnd_ - gapStart_; }
61 void moveGap(std::size_t offset);
62 void reserveGap(std::size_t needed);
63 // Iterator to byte @p k of the storage, gap included.
64 auto it(std::size_t k) noexcept { return data_.begin() + static_cast<std::ptrdiff_t>(k); }
65 auto it(std::size_t k) const noexcept { return data_.cbegin() + static_cast<std::ptrdiff_t>(k); }
66
67 std::vector<char> data_;
68 std::size_t gapStart_ = 0;
69 std::size_t gapEnd_ = 0;
70};

72/// Undo history limits, in bytes of recorded text. The defaults match
73/// GNU Emacs's `undo-limit` and `undo-strong-limit` (`tui.md` §8.7).
74struct undoLimits {
75 /// History is trimmed to this size after the change that crosses it.
76 std::size_t soft = 160'000;
77 /// History never exceeds this size, except for the most recent change.
78 std::size_t hard = 240'000;
79};
