54/// Engine construction settings.
55struct engineOptions {
56 /// Sample rate and block size.
57 prepareInfo audio;
58 /// Pipe table capacity.
59 pipeTable::capacity pipes;
60 /// Decodes image files for `imagePlayer` on the builder thread
61 /// (`primitives` does no file I/O, §1.2). Empty: images cannot load.
62 imageLoader loadImage = nullptr;
63 /// Decodes audio files for buffers on the builder thread. Empty:
64 /// buffers cannot load files.
65 sampleLoader loadSample = nullptr;
66 /// Bound on the memory of live buffers, in bytes (`samples.md` §5.6).
67 std::size_t bufferMemory = bufferPool::defaultLimit;
68 /// Times each node of the active plan into `plan::seconds`: two clock
69 /// reads a node a block, for offline profiles (`threads.md` §11).
70 bool profile = false;
71 /// The threads that run a block, the audio thread included
72 /// (`threads.md` §7): 1 runs every node on the audio thread in plan
73 /// order; more add workers that run the plan's tasks beside it. The
74 /// output is the same at every count.
75 int threads = 1;
76 /// Called first on each worker's own thread, to raise its priority,
77 /// for example; may be empty. It runs inside a noexcept function, so it
78 /// must not throw: a throw ends the process. It may still be running
79 /// when the engine's constructor returns, and the engine's destructor
80 /// waits for it to return. It must own what it captures, since the
81 /// engine's copy can outlive the objects in the caller's scope.
82 std::function<void()> workerStart = nullptr;
83 /// How long a worker that finds no task spins before it sleeps
84 /// (`threads.md` §6.5): longer wakes it sooner for the next block and
85 /// burns more of a core between blocks.
86 std::chrono::microseconds workerSpin = defaultWorkerSpin;
87};
