103/// The nodes of an execution order grouped into tasks (`threads.md` §5):
104/// a thread runs a task's nodes in order, start to finish, once every
105/// task it waits for has finished.
106struct taskList {
107 /// The nodes of every task, by position in the order: task by task,
108 /// each task's in execution order.
109 std::vector<std::uint32_t> nodes;
110 /// Where each task's nodes start in `nodes`; one entry more than there
111 /// are tasks.
112 std::vector<std::uint32_t> start{0};
113 /// The tasks that wait for each task, task by task, each task's sorted.
114 std::vector<std::uint32_t> successors;
115 /// Where each task's successors start in `successors`; one entry more
116 /// than there are tasks.
117 std::vector<std::uint32_t> successorStart{0};
118 /// The number of tasks each task waits for.
119 std::vector<std::uint32_t> waits;
120 /// The estimated work of a block run in order on one thread, in
121 /// nanoseconds a block: the sum of the nodes' estimated costs
122 /// (`nodeInfo::cost`, `threads.md` §5.2).
123 double serialWork = 0.0;
124 /// The estimated work of a block run as tasks, in nanoseconds a block:
125 /// the sum of the tasks' costs. A task costs its nodes' estimated costs
126 /// and a fixed cost for each dependency of one of them on a node of
127 /// another task, whose output its thread must read from another
128 /// processor's cache (`threads.md` §6.1.1).
129 double work = 0.0;
130 /// The estimated critical path, in nanoseconds a block: the costliest
131 /// path through the task graph by the tasks' costs. No number of
132 /// threads runs the tasks in less.
133 double criticalPath = 0.0;
134 /// True when the tasks run on the engine's threads: some level of the
135 /// task graph holds two tasks, which can run at once, and the
136 /// estimated saving at the engine's thread count (savingAt) reaches
137 /// `scheduleOptions::minimumParallelSaving`. The engine runs tasks
138 /// that are not parallel on the audio thread alone, in
139 /// `nodeSchedule::serialOrder`, waking no worker (`threads.md` §6.1.1).
140 bool parallel = false;
141 /// The most threads that run the tasks when they are parallel, the
142 /// audio thread included (`threads.md` §6.1.1). tasksOf sets it to the
143 /// fewest of the engine's threads whose estimated saving falls short of
144 /// the saving on all of them by less than
145 /// `scheduleOptions::minimumParallelSaving`, when that is fewer than
146 /// all: the threads past it would save less than that together, so
147 /// they sleep through the block instead of taking a share of the cores
148 /// from the threads that work. Otherwise, and in lists built by hand,
149 /// every thread of the pool runs them.
150 int threads = std::numeric_limits<int>::max();
151
152 /// The estimated saving, in nanoseconds a block, of running the tasks
153 /// on @p count threads instead of in order on one: `serialWork` less
154 /// the larger of `criticalPath` and `work` / @p count, the least time
155 /// @p count threads can take; 0 when that is negative, and for one
156 /// thread.
157 double savingAt(int count) const noexcept {
158 return count > 1 ? std::max(0.0, serialWork - std::max(criticalPath, work / count)) : 0.0;
159 }
160
161 /// The number of tasks.
162 std::size_t size() const noexcept { return start.size() - 1; }
163 /// The nodes of task @p t, which is less than size().
164 std::span<const std::uint32_t> nodesOf(std::size_t t) const noexcept {
165 return std::span(nodes).subspan(start[t], start[t + 1] - start[t]);
166 }
167 /// The tasks that wait for task @p t.
168 std::span<const std::uint32_t> successorsOf(std::size_t t) const noexcept {
169 return std::span(successors).subspan(successorStart[t], successorStart[t + 1] - successorStart[t]);
170 }
171};

183/// Items for each node of an execution order, node by node in one array.
184template <typename T>
185struct perNode {
186 /// Where each node's items start in `items`; one entry more than there
187 /// are nodes.
188 std::vector<std::uint32_t> start{0};
189 /// The items of every node, node by node.
190 std::vector<T> items;
191
192 /// The number of nodes.
193 std::size_t size() const noexcept { return start.size() - 1; }
194 /// The items of node @p i, which is less than size().
195 std::span<const T> of(std::size_t i) const noexcept {
196 return std::span(items).subspan(start[i], start[i + 1] - start[i]);
197 }

200/// For each node of an execution order, the earlier nodes it must run
201/// after (`threads.md` §4.1), by position in the order, each node's
202/// sorted. Every node a node must follow is reachable through the list,
203/// though not always directly: a node after two writers of a pipe lists
204/// only the second, which follows the first.
205using dependencyList = perNode<std::uint32_t>;
206
207/// The work a block does on its pipes (`threads.md` §6.8), placed so that
208/// most of it runs inside the tasks: each audio and event pipe of the
209/// table, and each private pipe, starts (zeroed or emptied for the block)
210/// just before the node that first uses it in the order runs, when that
211/// node writes it, and each audio pipe publishes its last samples just
212/// after the node that last writes it. The audio thread does the rest,
213/// before and after every node. Built on the builder thread with the
214/// order's schedule; the audio thread and the workers only read it.
215///
216/// A start in a node is race-free: the node is the first in the order to
217/// use the pipe, and every later user follows it through the dependencies
218/// (`threads.md` §4.1.2), in every order the dependencies admit. Feedback
219/// pipes, whose reads follow no writer, start and end on the audio thread.
220struct pipeWork {
221 /// The table's audio and event pipes each node starts before it runs.
222 perNode<metadata::pipeRef> starts;
223 /// The private pipes each node starts before it runs.
224 perNode<std::shared_ptr<aPipe>> privateStarts;
225 /// The table's audio pipes each node publishes after it runs
226 /// (aPipe::publish).
227 perNode<metadata::pipeRef> publishes;
228 /// For each node, the work it does on pipes, as bits: one byte the
229 /// runner reads for the many nodes that do none. `touchesPipes` when it
230 /// starts or publishes a table pipe or starts a private pipe without a
231 /// target; `touchesForwarding` when it starts a private pipe with one
232 /// (aPipe::hasTarget). A plan whose private pipes forward starts none
233 /// of those (`threads.md` §4.2.5), so its runner skips a node with
234 /// only the second bit without reading its pipes.
235 std::vector<std::uint8_t> touches;
236 /// The bit of `touches` for a table pipe's start or publish, or a
237 /// start of a private pipe without a target.
238 static constexpr std::uint8_t touchesPipes = 1;
239 /// The bit of `touches` for a start of a private pipe with a target.
240 static constexpr std::uint8_t touchesForwarding = 2;
241 /// The table's audio and event pipes the audio thread starts before any
242 /// node runs: feedback pipes, pipes no node writes, and pipes a node
243 /// reads before the first writes them.
244 std::vector<metadata::pipeRef> serialStarts;
245 /// The private pipes the audio thread starts before any node runs:
246 /// those no node of the order writes (in a crossfade, those of writers
247 /// the incoming plan adopted) and those a node reads before their
248 /// writer runs.
249 std::vector<std::shared_ptr<aPipe>> serialPrivateStarts;
250 /// The pipes the audio thread ends after every node has run
251 /// (pipeTable::endBlock): feedback audio and control pipes, and audio
252 /// pipes no node writes. The end of a control pipe that is not
253 /// feedback does nothing.
254 std::vector<metadata::pipeRef> serialEnds;
255};

257/// How the nodes of an execution order may run in parallel: their
258/// dependencies, their tasks, and the scheduler's counters for the tasks.
259struct nodeSchedule {

261 dependencyList dependencies;
262 /// The nodes grouped into tasks (`threads.md` §5).
263 taskList tasks;
264 /// The scheduler's counters for `tasks`.
265 taskCounters counters;
266 /// The order in which a thread that runs every node of a block alone
267 /// runs them (an engine of one thread, or tasks that are not parallel

269 /// `scheduleOptions::shuffleSeed`, a permutation of them that keeps
270 /// every dependency (`threads.md` §5.5).
271 std::vector<std::uint32_t> serialOrder;
272 /// The block's work on pipes, by node (`threads.md` §6.8).
273 pipeWork pipes;

276/// The least estimated cost of a task by default, in nanoseconds a block
277/// (`threads.md` §5.2, proposed): a task's bookkeeping, about a
278/// microsecond, then stays a few percent of its work.
279inline constexpr double defaultMinimumTaskCost = 20'000.0;

281/// The least estimated saving for which a schedule runs in parallel by
282/// default, in nanoseconds a block (`scheduleOptions::minimumParallelSaving`,
283/// `threads.md` §6.1.1). Calibrated on Release renders of every example on
284/// the development Mac at 2 threads: each example that renders slower in
285/// parallel is estimated to save nothing, and the one estimated to save
286/// least while rendering faster saves 7.5 µs.
287inline constexpr double defaultMinimumParallelSaving = 4'000.0;

289/// How the builder makes the schedules of plans and crossfades
290/// (`engineOptions::schedule`). The output is the same at every setting.
291struct scheduleOptions {
292 /// The least estimated cost of a task, in nanoseconds a block, not
293 /// negative (`threads.md` §5.2): the builder joins chains of nodes that
294 /// cannot reach each other into one task until it holds this much. At
295 /// 0 every chain is a task of its own, the finest partition, which
296 /// tests use so that independent nodes run on different threads.
297 double minimumTaskCost = defaultMinimumTaskCost;
298 /// The threads that run a block, the audio thread included: the
299 /// engine sets it to `engineOptions::threads`. At 1 no schedule is
300 /// parallel (taskList::parallel).
301 int threads = 1;
302 /// The least estimated saving a block, in nanoseconds, not negative,
303 /// for which tasks that can run at once run in parallel on `threads`
304 /// threads (taskList::savingAt, taskList::parallel); with less, the
305 /// block runs in order on the audio thread. At 0 every schedule whose
306 /// tasks can run at once is parallel, which tests use to run every
307 /// plan on the pool; an infinite one makes every schedule serial.
308 double minimumParallelSaving = defaultMinimumParallelSaving;
309 /// For testing that the dependencies capture every conflict
310 /// (`threads.md` §5.5, §9.1): when set, a block whose nodes one thread
311 /// runs alone (one thread, or tasks that are not parallel or do not fit
312 /// the pool) runs them in a random order that keeps every dependency
313 /// (`nodeSchedule::serialOrder`), drawn from this seed alone once for
314 /// each plan and each crossfade, instead of plan order. A node that
315 /// touches shared state it does not declare then renders differently.
316 std::optional<std::uint64_t> shuffleSeed = std::nullopt;
317};
