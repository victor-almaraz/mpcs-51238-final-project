18/// Enables flush-to-zero and denormals-are-zero for its scope and restores

20class denormalGuard {

22 denormalGuard() noexcept ESPRESSO_NONBLOCKING {
23#if defined(__SSE__) || defined(__x86_64__) || defined(_M_X64)
24 saved_ = _mm_getcsr();

26#elif defined(__aarch64__)

29#endif
