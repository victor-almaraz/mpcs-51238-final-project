129/// The rules a packet is decoded by.
130enum class oscRules : std::uint8_t {
131 /// The client and server's own messages (`protocol.md` §2.1, §2.2):
132 /// the type tags `i h f d s b`, padding of zeros, and nothing after
133 /// the last argument.
134 control,
135 /// Messages from other programs, such as those a piece sends to a page
136 /// (`protocol.md` §2.5, `io.md` §12.5): every type tag of OSC 1.1; a
137 /// message without a type tag string has no arguments, padding is
138 /// skipped unread, and bytes after the last argument are ignored.
139 lenient,
140};
141
142/// Decodes one OSC message by @p rules. Returns nothing for malformed
143/// input, bundles, and type tags the rules do not accept; it never throws
144/// on bad data (`protocol.md` §2.4).
145std::optional<message> decode(std::span<const std::uint8_t> packet, oscRules rules = oscRules::control);
