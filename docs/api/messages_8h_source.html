117/// Error codes carried by `/fail` (`protocol.md` §8).
118enum class errorCode : std::int32_t {
119 unknownAddress = 1, ///< No handler for the address
120 malformedArguments = 2, ///< Arguments missing, of the wrong type, or out of range
121 notConnected = 3, ///< The client has not completed `/hello`
122 versionMismatch = 4, ///< The client's metadata version differs from the server's
123 irIncomplete = 5, ///< The assembled IR size differs from `/ir.begin`
124 irChecksumMismatch = 6, ///< The assembled IR's CRC-32 differs from `/ir.begin`
125 irInvalid = 7, ///< The announcement or the IR is invalid, or the plan cannot be built
126 unknownPlan = 8, ///< No such plan, or it was already played
127 deviceError = 9, ///< A device failed
128 signalFault = 10, ///< A sink muted itself: non-finite samples
129 fileError = 11, ///< A buffer could not be saved to its file
130};

132/// True when replies to @p addr carry the request ID as their first
133/// argument.
134constexpr bool isReply(std::string_view addr) noexcept {
135 return addr == address::done || addr == address::fail || addr == address::helloOk ||
136 addr == address::statusReply || addr == address::irMissing || addr == address::controlValue ||
137 addr == address::synced;
138}

140/// What a `/fail` names in its trailing arguments (`protocol.md` §5).
141enum class failSubject : std::uint8_t {
142 none, ///< The plan or request as a whole
143 node, ///< One node
144 buffer, ///< One buffer
145};
146
147/// The name `/fail` carries for @p s: `node` or `buffer`; empty for none.
148constexpr std::string_view subjectName(failSubject s) noexcept {
149 switch (s) {
150 case failSubject::node: return "node";
151 case failSubject::buffer: return "buffer";
152 case failSubject::none: break;
153 }
154 return {};
155}
156
157/// The subject that @p name names; none for any other text.
158constexpr failSubject subjectNamed(std::string_view name) noexcept {
159 if (name == subjectName(failSubject::node)) return failSubject::node;
160 if (name == subjectName(failSubject::buffer)) return failSubject::buffer;
161 return failSubject::none;
162}

164/// Why a plan was refused (`protocol.md` §5): the message, and, when one
165/// node or buffer of the plan failed, which, so a client can find its
166/// source.
167struct planError {
168 /// What is wrong, as `/fail` reports it.
169 std::string message;
170 /// What failed; none for the plan as a whole.
171 failSubject subject = failSubject::none;
172 /// The node's or buffer's ID; empty for the plan as a whole.
173 std::string id;
174
175 /// An error about the plan as a whole.
176 static planError plain(std::string message) {
177 return {.message = std::move(message), .subject = failSubject::none, .id = {}};
178 }
179};
