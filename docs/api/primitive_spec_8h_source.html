25/// What a parameter's value means (`primitives.md` §7.4.1).
26enum class dimension : std::uint8_t {

29 frequency, ///< Hertz

34 text, ///< Free text, such as a device name or an OSC address
35 list, ///< A list of numbers
36 buffer, ///< A buffer, named by index (`samples.md` §12.1)
37 textList, ///< A list of strings, such as JSON field paths or request headers (`io.md` §9.1)

46/// A set of pipe kinds, as bit flags.
47using kindSet = std::uint8_t;
48
49/// The flag for @p kind.
50constexpr kindSet bit(pipeKind kind) noexcept {
51 return static_cast<kindSet>(1u << static_cast<unsigned>(kind));
52}
53
54/// No pipe kind: the parameter is static only.
55inline constexpr kindSet staticOnly = 0;
56/// Audio pipes.
57inline constexpr kindSet audioPipes = bit(pipeKind::audio);
58/// Control pipes.
59inline constexpr kindSet controlPipes = bit(pipeKind::control);
60/// Event pipes.
61inline constexpr kindSet eventPipes = bit(pipeKind::event);
62/// Audio or control pipes: a signal.
63inline constexpr kindSet signalPipes = audioPipes | controlPipes;
64/// True when @p set contains @p kind.
65constexpr bool contains(kindSet set, pipeKind kind) noexcept { return (set & bit(kind)) != 0; }

74/// One parameter of a primitive type.
75struct paramSpec {
76 /// Name, in kebab-case: `freq`, `max-time`.
77 std::string_view name;
78 /// Dimension of the value.
79 dimension dim = dimension::number;
80 /// Default, in canonical units, for numeric parameters.
81 double defaultValue = 0.0;
82 /// Smallest accepted static value.
83 double min = -1e9;
84 /// Largest accepted static value.
85 double max = 1e9;
86 /// Pipe kinds the parameter may be bound to.
87 kindSet bindable = staticOnly;
88 /// Accepted names, for `dimension::choice`.
89 std::span<const std::string_view> choices = noNames;
90 /// Default, for choice and text parameters.
91 std::string_view defaultText = noText;
92 /// True when a buffer parameter's node writes the buffer; false when it
93 /// only reads it (`samples.md` §5.7).
94 bool writes = false;
95 /// True when every node of the type must give the parameter, and give a
96 /// text parameter not empty: it has no default. checkConstraints()
97 /// reports a missing or empty one.
98 bool required = false;
99 /// True when a number parameter also accepts `:page`, the project's
100 /// page port, in the language (`io.md` §12.3). The IR holds the number.
101 bool pagePort = false;
102 /// True when a number parameter takes a value in any unit, held in the
103 /// canonical one: `osc-in`'s `:initial`, the first value of a pipe
104 /// that carries hertz, seconds, or a level (`io.md` §12.3).
105 bool anyUnit = false;
106 /// True when a text parameter's value is part of what a node is: a node
107 /// whose value differs from the outgoing node's does not take over its
108 /// state (`primitives.md` §9.2.2). `clip-player`'s `bank`.
109 bool identity = false;
110};

112/// One input or output port of a primitive type.
113struct portSpec {

134/// How a mapper operation (`primitives.md` §7.9) treats the dimensions of
135/// its operands: the input, then the parameters in table order.
136enum class operands : std::uint8_t {
137 none, ///< Not a mapper operation
138 plain, ///< Every operand is a plain number: exp, log, sin, cos, tanh, pow
139 keep, ///< The result has the input's dimension: floor, round
140 sign, ///< As `keep`, but a level is ambiguous: neg, abs
141 same, ///< Every operand has the input's dimension, as the result does: min, max, clamp, quantize
142 ranges, ///< The input shares a dimension with the first two parameters; the last two share the result's:
143 ///< scale, scaleExp
144};

146/// What a constraint requires of a node (`constraint`).
147enum class rule : std::uint8_t {
148 anyInput, ///< At least one of the input ports `ports` is bound
149 anyOutput, ///< At least one of the output ports `ports` is bound
150 entryPerPipe, ///< List parameter `param`, when it has entries, has one per pipe of input `ports[0]`
151 notEmpty, ///< List parameter `param` has an entry; an absent list is empty
152 channels, ///< Output `ports[0]`, when bound, has a pipe of at least `count` channels
153 audioWith, ///< When input `ports[0]` is bound, output `ports[1]` is an audio pipe
154 textWith, ///< When input `ports[0]` is bound, text or text list parameter `param` is not empty
155 powerOfTwo, ///< Numeric parameter `param`, given or by default, is a power of two
156 oneOf, ///< Numeric parameter `param`, given or by default, is one of `values`
157 scheme, ///< Text parameter `param`, when not empty, starts with one of `names`
158 atMost, ///< Text list parameter `param` has at most `count` entries
159 headers, ///< Each entry of text list parameter `param` is a request header, `Name: value`
160 entryPerChannel, ///< Text list parameter `param`, when it has entries, has one per channel of port
161 ///< `ports[0]`, an input or an output, when it is bound
162};
163
164/// A requirement that spans parameters and ports, which the engine's
165/// builder relies on.
166struct constraint {
167 /// What is required.
168 rule kind = rule::anyInput;
169 /// The parameter the rule names, if any.
170 std::string_view param = noText;
171 /// The ports the rule names.
172 std::span<const std::string_view> ports = noNames;
173 /// The channel count, for `rule::channels`; the most entries, for `rule::atMost`.
174 int count = 0;
175 /// The accepted values, for `rule::oneOf`.
176 std::span<const double> values = noValues;
177 /// The accepted prefixes, for `rule::scheme`.
178 std::span<const std::string_view> names = noNames;
179};
180
181/// No constraints: the default of `primitiveSpec::constraints`.
182inline constexpr std::span<const constraint> noConstraints{};

184/// The table entry for one primitive type.
185struct primitiveSpec {
186 /// IR type name: `sineOsc`.
187 std::string_view type;
188 /// Source, sink, or generator.
189 role kind = role::generator;
190 /// Parameters.
191 std::span<const paramSpec> params;
192 /// Input ports.
193 std::span<const portSpec> inputs;
194 /// Output ports.
195 std::span<const portSpec> outputs;
196 /// Requirements across parameters and ports.
197 std::span<const constraint> constraints = noConstraints;
198 /// Whether the type is a mapper operation, and how it treats dimensions.
199 /// A mapper operation reads `in` and takes its operands as its
200 /// parameters, in order.
201 operands mapper = operands::none;
202
203 /// The parameter named @p name, if there is one.
204 optionalRef<paramSpec> param(std::string_view name) const noexcept;
205 /// The input port named @p name, if there is one.
206 optionalRef<portSpec> input(std::string_view name) const noexcept;
207 /// The output port named @p name, if there is one.
208 optionalRef<portSpec> output(std::string_view name) const noexcept;
209};
