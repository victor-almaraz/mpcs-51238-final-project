23/// Decoded audio at the engine's sample rate: what a sample loader returns
24/// (`samples.md` §3.3).
25struct decodedAudio {
26 /// One vector of samples per channel, all of one length.
27 std::vector<std::vector<float>> channels;
28
29 /// The length, in frames: that of the first channel, or 0.
30 std::size_t frames() const noexcept { return channels.empty() ? 0 : channels.front().size(); }
31};

33/// Decodes the audio file at @p path and resamples it to @p sampleRate.
34/// Returns null, with @p error set, when it cannot. Runs on the builder
35/// thread (`primitives` does no file I/O). The result is shared, so a
36/// loader may cache it.
37using sampleLoader = std::function<std::shared_ptr<const decodedAudio>(
38 std::string_view path, double sampleRate, std::string& error)>;

40/// A block of audio memory of a fixed number of channels and frames,
41/// planar and zeroed when made (`samples.md` §2). Allocated by the
42/// builder, written and read on the audio thread, freed with the last
43/// plan that holds it on the control thread.
44class sampleBuffer {
45public:
46 /// A silent buffer of @p channels channels, 1 to `metadata::maxChannels`,
47 /// and @p frames frames.
48 sampleBuffer(int channels, std::size_t frames);
49
50 /// Channel count.
51 int channels() const noexcept { return channels_; }
52 /// Length, in frames.
53 std::size_t frames() const noexcept { return frames_; }
54 /// Memory held, in bytes.
55 std::size_t bytes() const noexcept { return samples_.size() * sizeof(float); }
56
57 /// The samples of channel @p c, from 0.
58 std::span<float> channel(int c) noexcept;
59 /// The samples of channel @p c, from 0.
60 std::span<const float> channel(int c) const noexcept;
61
62 /// Copies @p audio in from the start, mapping its channels to the
63 /// buffer's (`samples.md` §4.4). A longer file is cut; a shorter one
64 /// leaves the rest silent.
65 void fill(const decodedAudio& audio) noexcept;
66
67private:
68 int channels_;
69 std::size_t frames_;
70 std::vector<float> samples_; // channel 0's frames, then channel 1's
71};
