110/// Validates and answers client messages for one server (`protocol.md` §3–§6).
111///
112/// The session remembers each client that completed `/hello`. UDP gives no
113/// notice when a client goes away, and an idle editor sends nothing while
114/// it still receives notifications, so clients are not expired by time.
115/// Instead, a protocol::client sends `/notify 0` when it is destroyed, and
116/// at most `maxClients` clients are kept: a `/hello` from a new client
117/// beyond that forgets the client heard from least recently.
118class serverSession {
119public:
120 /// Creates a session that reports @p info and drives @p e.
121 serverSession(serverInfo info, engine& e);
122
123 /// Decodes and handles one datagram. Undecodable datagrams are
124 /// dropped and counted (`protocol.md` §2.4).
125 std::vector<outgoing> handleDatagram(std::span<const std::uint8_t> bytes, endpoint from);
126
127 /// Handles one decoded message from @p from.
128 std::vector<outgoing> handle(const message& m, endpoint from);
129
130 /// @p m addressed to every client that sent `/notify 1`.
131 std::vector<outgoing> broadcast(const message& m) const;
132
133 /// The periodic `/counters` message for subscribed clients: missed
134 /// deadlines, xruns, drops, overflows, load, tempo, beat, and meter.
135 std::vector<outgoing> countersMessages() const;
136
137 /// Counts @p n datagrams that did not decode or could not be sent
138 /// (`protocol.md` §2.4). They are reported in the drops counter.
139 void countDropped(std::size_t n = 1) noexcept { dropped_ += n; }
140
141 /// Datagrams dropped because they did not decode or could not be
142 /// sent, and IR transfers abandoned to stay within the limits.
143 std::size_t droppedDatagrams() const noexcept { return dropped_ + ir_.abandoned(); }
144

146 serverCounters counters() const;

148 /// True when @p client completed `/hello`.
149 bool isConnected(endpoint client) const { return clients_.contains(client); }

151 /// Number of clients that completed `/hello` and are remembered.
152 std::size_t clientCount() const noexcept { return clients_.size(); }
