19/// Owns all pipes. Allocates everything at construction (§6.2).
20class pipeTable {
21public:
22 /// Pipe counts per type and event pipe sizes.
23 struct capacity {
24 /// Audio pipes.
25 int audio = 256;
26 /// Control pipes, including the tempo pipe at index 0.
27 int control = 1024;
28 /// Event pipes.
29 int event = 128;
30 /// Events per event pipe per block.
31 int eventsPerPipe = 256;
32 /// Pool bytes per event pipe per block, for OSC messages.
33 int poolBytes = 8192;
34 /// Video pipes, each a frame of the engine's frame size. None by
35 /// default: a frame is megabytes, and only a server with a window
36 /// or an offline frame render needs them.
37 int video = 0;
38 };
39
40 /// Allocates every pipe for blocks of `info.blockSize`. The tempo
41 /// pipe starts at 120 BPM.
42 pipeTable(const capacity& limits, const prepareInfo& info);

82private:
83 // Shared with every event pipe, so a pipe that outlives the table
84 // still counts into live memory.
85 ePipe::counter overflows_ = std::make_shared<std::atomic<std::uint64_t>>(0);
86 // Shared with the primitives bound to them, so a primitive's pipes
