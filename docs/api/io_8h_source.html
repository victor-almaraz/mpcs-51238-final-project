53/// The numbers of one serial line, or of one JSON document, with their
54/// time on the device clock (`io.md` §3.1). Fixed-size so ports do not
55/// allocate; the I/O threads parse and format, the audio thread never does.
56struct dataFrame {
57 /// Receive or send time, in seconds on the device clock.
58 double time = 0.0;
59 /// Values used, up to `metadata::maxChannels`.
60 std::uint8_t count = 0;
61 /// Bit *i* set: value *i* is missing, as a JSON field can be (§8.2),
62 /// and its channel keeps its value.
63 std::uint8_t missing = 0;
64 /// The values; value *i* is channel *i*.
65 std::array<float, metadata::maxChannels> values{};
66
67 /// True when value @p i is used and not missing.
68 constexpr bool has(std::size_t i) const noexcept {
69 return i < count && i < values.size() && (missing & (1U << i)) == 0;
70 }
71};

105/// A queue between one primitive and the I/O layer. Shared by the
106/// primitive and its plan; the hub keeps only a weak reference, so a port
107/// is gone when its plan is. Input ports accept data only while active,
108/// so a plan that is loaded but not yet playing does not accumulate stale
109/// messages. An input port names the node that reads it: the hub delivers
110/// each message to one active port per node, so an adopting node and the
111/// node it replaces, both active during a swap, never receive it twice.
112class ioPort {
113public:
114 virtual ~ioPort() = default;
115
116 ioPort(const ioPort&) = delete;
117 ioPort& operator=(const ioPort&) = delete;
118
119 /// What the port connects to.
120 portKind kind() const noexcept { return kind_; }
121 /// Node ID of the node reading an input port or writing an output
122 /// port; empty for MIDI and OSC outputs.
123 const std::string& owner() const noexcept { return owner_; }
124 /// MIDI device name pattern; empty matches every device.
125 const std::string& device() const noexcept { return device_; }
126 /// UDP port, for OSC ports.
127 std::uint16_t udpPort() const noexcept { return udpPort_; }
128
129 /// Activates or deactivates the port (audio thread, at a swap).
130 void setActive(bool on) noexcept ESPRESSO_NONBLOCKING { active_.store(on, std::memory_order_seq_cst); }
131 /// True while the port's plan is running.
132 bool active() const noexcept { return active_.load(std::memory_order_seq_cst); }
133
134 /// A delivery in progress (hub thread), from before it reads whether
135 /// the port is active until after it pushes. The audio thread may
136 /// deactivate the port in between; closed() tells it when no delivery
137 /// can still push.
138 class delivery {
139 public:
140 /// Starts a delivery to @p port.
141 explicit delivery(ioPort& port) noexcept : port_(port) {
142 port_.delivering_.fetch_add(1, std::memory_order_seq_cst);
143 }
144 ~delivery() { port_.delivering_.fetch_sub(1, std::memory_order_seq_cst); }
145
146 delivery(const delivery&) = delete;
147 delivery& operator=(const delivery&) = delete;
148 delivery(delivery&&) = delete;
149 delivery& operator=(delivery&&) = delete;
150
151 /// True when the port was active once the delivery had started.
152 bool admitted() const noexcept { return port_.active(); }
153
154 private:
155 ioPort& port_;
156 };
157
158 /// True when no delivery can push into the port any more: it is
159 /// inactive and none is in progress. A delivery that starts later
160 /// reads the port inactive and pushes nothing (sequentially consistent
161 /// order).
162 bool closed() const noexcept ESPRESSO_NONBLOCKING {
163 return !active() && delivering_.load(std::memory_order_seq_cst) == 0;
164 }
165
166protected:
167 /// A port of @p kind for @p device or @p udpPort, read by node @p owner.
168 ioPort(portKind kind, std::string owner, std::string device, std::uint16_t udpPort)
169 : kind_(kind), owner_(std::move(owner)), device_(std::move(device)), udpPort_(udpPort) {}
170
171private:
172 portKind kind_ = portKind::midiIn;
173 std::string owner_;
174 std::string device_;
175 std::uint16_t udpPort_ = 0;
176 std::atomic<bool> active_{false};
177 std::atomic<int> delivering_{0};
178};
