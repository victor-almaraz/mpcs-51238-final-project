23/// Reads and writes files. Has no terminal dependency (`tui.md` §3.1).
24class fileStore {
25public:
26 /// Reads @p path as UTF-8. A file that is not valid UTF-8 is
27 /// rejected with an error naming the file (`tui.md` §7.1).
28 static readFileResult read(const std::filesystem::path& path);
29
30 /// Writes @p text to a temporary file in the target's directory,
31 /// flushes it, and renames it over @p path. When @p path is a
32 /// symbolic link, the file it names is replaced and the link is kept.
33 /// On failure the original file is unchanged and the returned message
34 /// describes the error (`tui.md` §7.2).
35 static std::optional<std::string> writeAtomic(const std::filesystem::path& path, const std::string& text);
36
37 /// Creates an empty file at @p path, with the directories above it that
38 /// are missing. Fails, and changes nothing, when something by that name
39 /// exists; the returned message names it (`patterns.md` §20.7).
40 static std::optional<std::string> create(const std::filesystem::path& path);
41
42 /// Renames @p from to @p to, a file or a directory, creating the
43 /// directories above @p to that are missing. Fails, and changes
44 /// nothing, when @p to exists or @p from does not.
45 static std::optional<std::string> rename(const std::filesystem::path& from,
46 const std::filesystem::path& to);
47
48 /// When @p path was last written, as an opaque token that changes when
49 /// the file does (its modification time, in the file clock's ticks);
50 /// nothing when it cannot be read. An editor keeps the token it read
51 /// and writes only while the file still has it (`patterns.md` §13).
52 static std::optional<std::string> modified(const std::filesystem::path& path);
53
54 /// Lists regular files under @p root, sorted, relative to it.
55 /// Hidden entries (names starting with `.`) and `build` directories
56 /// are skipped. Symbolic links are listed by their own names, and
57 /// linked directories are descended into, except a link to a
58 /// directory that contains it, which would repeat forever.
59 /// At most @p limit files are listed: the walk stops at the next one, so
60 /// a caller that gets @p limit + 1 knows the folder holds more.
61 static std::vector<std::filesystem::path>
62 listProject(const std::filesystem::path& root,
63 std::size_t limit = std::numeric_limits<std::size_t>::max());
64};
