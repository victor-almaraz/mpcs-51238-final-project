23/// What a parameter's value means (`primitives.md` §7.4.1).
24enum class dimension : std::uint8_t {
25 number, ///< A plain number without a unit
26 time, ///< Seconds
27 frequency, ///< Hertz
28 level, ///< Linear amplitude
29 interval, ///< Semitones
30 beats, ///< Musical time, in beats
31 choice, ///< One name from a fixed list
32 text, ///< Free text, such as a device name or an OSC address
33 list, ///< A list of numbers
34 color, ///< RGBA, each component 0 to 1 (straight alpha); `none` where allowed
35};

37/// Lowercase name of @p d, for messages.
38std::string_view toString(dimension d) noexcept;

40/// A set of pipe kinds, as bit flags.
41using kindSet = std::uint8_t;
42
43/// The flag for @p kind.
44constexpr kindSet bit(pipeKind kind) noexcept {
45 return static_cast<kindSet>(1u << static_cast<unsigned>(kind));
46}
47
48/// No pipe kind: the parameter is static only.
49inline constexpr kindSet staticOnly = 0;
50/// Audio pipes.
51inline constexpr kindSet audioPipes = bit(pipeKind::audio);
52/// Control pipes.
53inline constexpr kindSet controlPipes = bit(pipeKind::control);
54/// Event pipes.
55inline constexpr kindSet eventPipes = bit(pipeKind::event);
56/// Audio or control pipes: a signal.
57inline constexpr kindSet signalPipes = audioPipes | controlPipes;
58/// Video pipes.
59inline constexpr kindSet videoPipes = bit(pipeKind::video);
60
61/// An RGBA color, each component 0 to 1, straight alpha.
62using rgba = std::array<double, 4>;
63
64/// Channels of a control pipe that carries a color: R, G, B, A.
65inline constexpr int colorChannels = 4;
66
67/// True when @p x is a color component: 0 to 1. NaN is not.
68constexpr bool isColorComponent(double x) noexcept { return x >= 0.0 && x <= 1.0; }
69
70/// True when @p c is a color: colorChannels components, each 0 to 1.
71constexpr bool isColor(std::span<const double> c) noexcept {
72 return c.size() == colorChannels && std::ranges::all_of(c, isColorComponent);
73}
74
75/// True when @p set contains @p kind.
76constexpr bool contains(kindSet set, pipeKind kind) noexcept { return (set & bit(kind)) != 0; }

78/// No text: the default of an optional text field.
79inline constexpr std::string_view noText{};

83/// One parameter of a primitive type.
84struct paramSpec {
85 /// Name, in kebab-case: `freq`, `max-time`.
86 std::string_view name;
87 /// Dimension of the value.
88 dimension dim = dimension::number;
89 /// Default, in canonical units, for numeric parameters.
90 double defaultValue = 0.0;
91 /// Smallest accepted static value.
92 double min = -1e9;
93 /// Largest accepted static value.
94 double max = 1e9;
95 /// Pipe kinds the parameter may be bound to.
96 kindSet bindable = staticOnly;
97 /// Accepted names, for `dimension::choice`.
98 std::span<const std::string_view> choices = noNames;
99 /// Default, for choice and text parameters.
100 std::string_view defaultText = noText;
101 /// Default, for color parameters.
102 rgba defaultColor = {0.0, 0.0, 0.0, 1.0};
103 /// True when a color parameter accepts `none` (no fill, no stroke).
104 bool noneAllowed = false;
105};

107/// One input or output port of a primitive type.
108struct portSpec {
109 /// Name: `in`, `out`, `gate`.
110 std::string_view name;
111 /// Pipe kinds the port accepts.
112 kindSet kinds = audioPipes;
113 /// True when the port must be bound.
114 bool required = false;
115 /// True when the port accepts more than one pipe.
116 bool multiple = false;
117};

119/// What a primitive does with external data (`primitives.md` §7).
120enum class role : std::uint8_t {
121 source, ///< Moves external data into pipes
122 sink, ///< Moves pipe data out of the graph
123 generator, ///< Produces signals from parameters and pipes
124};
125
126/// How a mapper operation (`primitives.md` §7.9) treats the dimensions of
127/// its operands: the input, then the parameters in table order.
128enum class operands : std::uint8_t {
129 none, ///< Not a mapper operation
130 plain, ///< Every operand is a plain number: exp, log, sin, cos, tanh, pow
131 keep, ///< The result has the input's dimension: floor, round
132 sign, ///< As `keep`, but a level is ambiguous: neg, abs
133 same, ///< Every operand has the input's dimension, as the result does: min, max, clamp, quantize
134 ranges, ///< The input shares a dimension with the first two parameters; the last two share the result's:
135 ///< scale, scaleExp
136};

138/// What a constraint requires of a node (`constraint`).
139enum class rule : std::uint8_t {
140 anyInput, ///< At least one of the input ports `ports` is bound
141 anyOutput, ///< At least one of the output ports `ports` is bound
142 entryPerPipe, ///< List parameter `param`, when it has entries, has one per pipe of input `ports[0]`
143 notEmpty, ///< List parameter `param` has an entry; an absent list is empty
144 channels, ///< Output `ports[0]`, when bound, has a pipe of at least `count` channels
145 audioWith, ///< When input `ports[0]` is bound, output `ports[1]` is an audio pipe
146 textWith, ///< When input `ports[0]` is bound, text parameter `param` is not empty
147};
148
149/// A requirement that spans parameters and ports, which the engine's
150/// builder relies on.
151struct constraint {
152 /// What is required.
153 rule kind = rule::anyInput;
154 /// The parameter the rule names, if any.
155 std::string_view param = noText;
156 /// The ports the rule names.
157 std::span<const std::string_view> ports = noNames;
158 /// The channel count, for `rule::channels`.
159 int count = 0;
160};
161
162/// No constraints: the default of `primitiveSpec::constraints`.
163inline constexpr std::span<const constraint> noConstraints{};
164
165/// The table entry for one primitive type.
166struct primitiveSpec {
167 /// IR type name: `sineOsc`.
168 std::string_view type;
169 /// Source, sink, or generator.
170 role kind = role::generator;
171 /// Parameters.
172 std::span<const paramSpec> params;
173 /// Input ports.
174 std::span<const portSpec> inputs;
175 /// Output ports.
176 std::span<const portSpec> outputs;
177 /// Requirements across parameters and ports.
178 std::span<const constraint> constraints = noConstraints;
179 /// Whether the type is a mapper operation, and how it treats dimensions.
180 /// A mapper operation reads `in` and takes its operands as its
181 /// parameters, in order.
182 operands mapper = operands::none;
183
184 /// The parameter named @p name, if there is one.
185 optionalRef<paramSpec> param(std::string_view name) const noexcept;
186 /// The input port named @p name, if there is one.
187 optionalRef<portSpec> input(std::string_view name) const noexcept;
188 /// The output port named @p name, if there is one.
189 optionalRef<portSpec> output(std::string_view name) const noexcept;
190};
