15/// Kind of a lexical token.
16enum class tokenKind : std::uint8_t {
17 leftParen, ///< `(`
18 rightParen, ///< `)`
19 leftBracket, ///< `[`
20 rightBracket, ///< `]`
21 quote, ///< `'`

28 keyword, ///< `:freq`
29 symbol, ///< `sine-osc`
30 lineComment, ///< `; note`
31 blockComment, ///< `#| ... |#`, nesting
32 datumComment, ///< `#;`, which comments out the next datum
33 error, ///< Malformed input; see token::error
34};

36/// Reason a token is malformed.
37enum class lexError : std::uint8_t {

39 unterminatedString,

73/// A token. Text is not copied: offset and length index the source.
74struct token {
75 /// Kind of token.
76 tokenKind kind;
77 /// Why the token is malformed, for tokenKind::error; lexError::none otherwise.
78 lexError error = lexError::none;
79 std::uint32_t offset = 0; ///< Byte offset of the first byte
80 std::uint32_t length = 0; ///< Length in bytes
81 std::uint32_t line = 1; ///< 1-based line
82 std::uint32_t column = 1; ///< 1-based column, in code points
83 std::uint32_t numericLength = 0; ///< For unit literals: bytes of the numeric part
84
85 /// Returns the token's text within @p source.
86 std::string_view text(std::string_view source) const noexcept { return source.substr(offset, length); }
87};
88
89/// Result of strict lexing.
90struct lexResult {
91 std::vector<token> tokens; ///< Tokens before the first error, or all tokens
92 std::optional<token> failure; ///< The first malformed token, if any
93};
94
95/// Lexes @p source in strict mode, for compilation. Stops at the first
96/// malformed token and reports it in lexResult::failure.
97lexResult lexStrict(std::string_view source);
98
99/// Lexes @p source in tolerant mode, for editor highlighting. Never
100/// fails: malformed input becomes tokens of kind tokenKind::error, and
101/// lexing continues after them. Whitespace produces no tokens.
102std::vector<token> lexTolerant(std::string_view source);
