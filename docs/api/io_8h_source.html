50/// An OSC message with its time on the device clock.
51struct oscPacket {
52 /// Receive or send time, in seconds on the device clock.
53 double time = 0.0;
54 /// Bytes used.
55 std::uint16_t size = 0;
56 /// The encoded OSC message.
57 std::array<std::byte, maxOscBytes> bytes{};
58
59 /// The packet holding @p message at @p time; nothing when it is larger
60 /// than maxOscBytes.
61 static std::optional<oscPacket> of(std::span<const std::uint8_t> message, double time = 0.0) noexcept {
62 if (message.size() > maxOscBytes) return std::nullopt;
63 oscPacket p;
64 p.time = time;
65 p.size = static_cast<std::uint16_t>(message.size());
66 std::ranges::transform(message, p.bytes.begin(), [](std::uint8_t b) { return std::byte{b}; });
67 return p;
68 }
69 /// The bytes it holds: the first `size`, at most all of them.
70 std::span<const std::byte> used() const noexcept {
71 return std::span(bytes).first(std::min<std::size_t>(size, bytes.size()));
72 }
73};

75/// A single-producer, single-consumer queue of OSC packets that keeps of
76/// each only its time, its size, and the bytes it uses, one after another
77/// in a ring (`primitives.md` §7.3.5). A message of a few dozen bytes
78/// takes a few dozen, where a slot of oscPacket takes over 1 KB: a piece
79/// of hundreds of OSC inputs, each with its own port, holds megabytes of
80/// queues rather than a hundred. The ring is written at construction, so
81/// neither end takes a page fault on its first use.
82// NOLINTNEXTLINE(clang-analyzer-optin.performance.Padding): cache-line separation
83class oscQueue {
84public:
85 /// A ring of at least @p bytes bytes, rounded up to a power of two.
86 explicit oscQueue(std::size_t bytes);
87
88 ~oscQueue() = default;
89 oscQueue(const oscQueue&) = delete;
90 oscQueue& operator=(const oscQueue&) = delete;
91 oscQueue(oscQueue&&) = delete;
92 oscQueue& operator=(oscQueue&&) = delete;
93
94 /// Producer: appends @p packet's time, size, and used bytes (at most
95 /// maxOscBytes, oscPacket::used). Returns false when the ring has no
96 /// room for them.
97 [[nodiscard]] bool push(const oscPacket& packet) noexcept ESPRESSO_NONBLOCKING;
98 /// Consumer: moves the oldest packet's time, size, and bytes into
99 /// @p out; its bytes past the size are left as they were. Returns
100 /// false when the queue is empty.
101 [[nodiscard]] bool pop(oscPacket& out) noexcept ESPRESSO_NONBLOCKING;
102 /// Consumer: true when there is nothing to pop.
103 bool empty() const noexcept ESPRESSO_NONBLOCKING;
104 /// The most packets the ring can hold at once, all empty: a bound for
105 /// a loop that drains it.
106 std::size_t capacity() const noexcept ESPRESSO_NONBLOCKING { return ring_.size() / header; }
107
108private:
109 // A packet's time and size, ahead of its bytes.
110 static constexpr std::size_t header = sizeof(double) + sizeof(std::uint16_t);
111
112 // Copies @p from into the ring from position @p at, wrapping at its end.
113 void put(std::size_t at, std::span<const std::byte> from) noexcept ESPRESSO_NONBLOCKING;
114 // Copies the ring from position @p at into @p to, wrapping at its end.
115 void get(std::size_t at, std::span<std::byte> to) const noexcept ESPRESSO_NONBLOCKING;
116
117 std::vector<std::byte> ring_;
118 std::size_t mask_;
119 static_assert(realtimeAtomic<std::size_t>);
120 // Positions are counts of bytes since construction: head_ the first
121 // byte not yet popped, tail_ the first not yet pushed.
122 alignas(cacheLine) std::atomic<std::size_t> head_{0};
123 alignas(cacheLine) std::atomic<std::size_t> tail_{0};
124};

131/// The numbers of one serial line, or of one JSON document, with their
132/// time on the device clock (`io.md` §3.1). Fixed-size so ports do not
133/// allocate; the I/O threads parse and format, the thread that runs the
134/// node never does.
135struct dataFrame {
136 /// Receive or send time, in seconds on the device clock.
137 double time = 0.0;
138 /// Values used, up to `metadata::maxChannels`.
139 std::uint8_t count = 0;
140 /// Bit *i* set: value *i* is missing, as a JSON field can be (§8.2),
141 /// and its channel keeps its value.
142 std::uint8_t missing = 0;
143 /// The values; value *i* is channel *i*.
144 std::array<float, metadata::maxChannels> values{};
145
146 /// True when value @p i is used and not missing.
147 constexpr bool has(std::size_t i) const noexcept {
148 return i < count && i < values.size() && (missing & (1U << i)) == 0;
149 }
150 /// The values used: the first `count`, at most all of them.
151 constexpr std::span<const float> used() const noexcept {
152 return std::span(values).first(std::min<std::size_t>(count, values.size()));
153 }
154};
155// The missing mask has a bit per channel.
156static_assert(metadata::maxChannels <= 8);

213/// A queue between one primitive and the I/O layer. Shared by the
214/// primitive and its plan; the hub keeps only a weak reference, so a port
215/// is gone when its plan is. Input ports accept data only while active,
216/// so a plan that is loaded but not yet playing does not accumulate stale
217/// messages. An input port names the node that reads it: the hub delivers
218/// each message to one active port per node, so an adopting node and the
219/// node it replaces, both active during a swap, never receive it twice.
220class ioPort {
221public:
222 /// Advances the hub's generation: the port is gone.
223 virtual ~ioPort() { generation_->fetch_add(1, std::memory_order_seq_cst); }
224
225 ioPort(const ioPort&) = delete;
226 ioPort& operator=(const ioPort&) = delete;
227
228 /// What the port connects to.
229 portKind kind() const noexcept { return kind_; }
230 /// Node ID of the node reading an input port or writing an output
231 /// port; empty for MIDI and OSC outputs.
232 const std::string& owner() const noexcept { return owner_; }
233 /// MIDI device name pattern; empty matches every device.
234 const std::string& device() const noexcept { return device_; }
235 /// UDP port, for OSC ports.
236 std::uint16_t udpPort() const noexcept { return udpPort_; }
237
238 /// Activates or deactivates the port (audio thread, at a swap), then
239 /// advances the hub's generation.
240 void setActive(bool on) noexcept ESPRESSO_NONBLOCKING {
241 active_.store(on, std::memory_order_seq_cst);
242 generation_->fetch_add(1, std::memory_order_seq_cst);
243 }
244 /// True while the port's plan is running.
245 bool active() const noexcept { return active_.load(std::memory_order_seq_cst); }

247 /// A delivery in progress (hub thread), from before it reads whether
248 /// the port is active until after it pushes. The audio thread may
249 /// deactivate the port in between; closed() tells it when no delivery

269 };

271 /// True when no delivery can push into the port any more: it is
272 /// inactive and none is in progress. A delivery that starts later
273 /// reads the port inactive and pushes nothing (sequentially consistent
274 /// order).
275 bool closed() const noexcept ESPRESSO_NONBLOCKING {
276 return !active() && delivering_.load(std::memory_order_seq_cst) == 0;
277 }

279protected:
280 /// A port of @p kind for @p device or @p udpPort, read by node @p owner,
281 /// made by the hub that gave @p key.

283 : kind_(kind),
284 owner_(std::move(owner)),
285 device_(std::move(device)),
286 udpPort_(udpPort),
287 generation_(std::move(key.generation_)) {}
288
289private:
290 portKind kind_ = portKind::midiIn;
291 std::string owner_;
292 std::string device_;
293 std::uint16_t udpPort_ = 0;
294 // The hub's generation, which may outlive the hub.
295 std::shared_ptr<std::atomic<std::uint64_t>> generation_;
296 static_assert(realtimeAtomic<bool> && realtimeAtomic<int> && realtimeAtomic<std::uint64_t>);
297 std::atomic<bool> active_{false};
298 std::atomic<int> delivering_{0};
