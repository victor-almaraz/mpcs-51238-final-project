18/// What an element of a layout is.
19enum class pageRole : std::uint8_t {
20 group, ///< Holds other elements, of the kinds its entry names
21 sends, ///< A control: sends to its address, which an `oscIn` on the page port reads
22 shows, ///< A display: shows what an `oscOut` on the page port sends to its address
23};

25/// The table entry for one kind of layout element.
26struct pageKindSpec {
27 /// The kind, as the IR names it: `section`, `knob`.
28 std::string_view kind;
29 /// Group, control, or display.
30 pageRole role = pageRole::group;
31 /// The custom element that draws it: `es-section`, `es-knob`.
32 std::string_view element;
33 /// Its parameters, besides the placement every element but the root
34 /// takes (`placementParams()`).
35 std::span<const paramSpec> params;
36 /// For a group, the kinds it holds.
37 std::span<const std::string_view> children = noNames;
38 /// For a control or a display, the address it takes when it gives none;
39 /// empty when it MUST give one.
40 std::string_view defaultAddress = noText;
41
42 /// The parameter named @p name, among its own and the placement
43 /// parameters, if there is one.
44 optionalRef<paramSpec> param(std::string_view name) const noexcept;
45};
