78/// A queue between one primitive and the I/O layer. Shared by the
79/// primitive and its plan; the hub keeps only a weak reference, so a port
80/// is gone when its plan is. Input ports accept data only while active,
81/// so a plan that is loaded but not yet playing does not accumulate stale
82/// messages. An input port names the node that reads it: the hub delivers
83/// each message to one active port per node, so an adopting node and the
84/// node it replaces, both active during a swap, never receive it twice.
85class ioPort {
86public:
87 virtual ~ioPort() = default;
88
89 ioPort(const ioPort&) = delete;
90 ioPort& operator=(const ioPort&) = delete;
91
92 /// What the port connects to.
93 portKind kind() const noexcept { return kind_; }
94 /// Node ID of the source reading an input port; empty for outputs.
95 const std::string& owner() const noexcept { return owner_; }
96 /// MIDI device name pattern; empty matches every device.
97 const std::string& device() const noexcept { return device_; }
98 /// UDP port, for OSC ports.
99 std::uint16_t udpPort() const noexcept { return udpPort_; }
100
101 /// Activates or deactivates the port (audio thread, at a swap).
102 void setActive(bool on) noexcept ESPRESSO_NONBLOCKING { active_.store(on, std::memory_order_seq_cst); }
103 /// True while the port's plan is running.
104 bool active() const noexcept { return active_.load(std::memory_order_seq_cst); }
105
106 /// A delivery in progress (hub thread), from before it reads whether
107 /// the port is active until after it pushes. The audio thread may
108 /// deactivate the port in between; closed() tells it when no delivery
109 /// can still push.
110 class delivery {
111 public:
112 /// Starts a delivery to @p port.
113 explicit delivery(ioPort& port) noexcept : port_(port) {
114 port_.delivering_.fetch_add(1, std::memory_order_seq_cst);
115 }
116 ~delivery() { port_.delivering_.fetch_sub(1, std::memory_order_seq_cst); }
117
118 delivery(const delivery&) = delete;
119 delivery& operator=(const delivery&) = delete;
120 delivery(delivery&&) = delete;
121 delivery& operator=(delivery&&) = delete;
122
123 /// True when the port was active once the delivery had started.
124 bool admitted() const noexcept { return port_.active(); }
125
126 private:
127 ioPort& port_;
128 };
129
130 /// True when no delivery can push into the port any more: it is
131 /// inactive and none is in progress. A delivery that starts later
132 /// reads the port inactive and pushes nothing (sequentially consistent
133 /// order).
134 bool closed() const noexcept ESPRESSO_NONBLOCKING {
135 return !active() && delivering_.load(std::memory_order_seq_cst) == 0;
136 }
137
138protected:
139 /// A port of @p kind for @p device or @p udpPort, read by node @p owner.
140 ioPort(portKind kind, std::string owner, std::string device, std::uint16_t udpPort)
141 : kind_(kind), owner_(std::move(owner)), device_(std::move(device)), udpPort_(udpPort) {}
142
143private:
144 portKind kind_ = portKind::midiIn;
145 std::string owner_;
146 std::string device_;
147 std::uint16_t udpPort_ = 0;
148 std::atomic<bool> active_{false};
149 std::atomic<int> delivering_{0};
150};

176/// A buffer's snapshot on its way to a file (`samples.md` §18), from a
177/// `bufferSave` node to the control thread. The node copies the buffer in
178/// while no save is pending and marks the snapshot ready; the control
179/// thread writes the file and frees the port for the next save.
180class savePort final : public ioPort {
181public:
182 /// Made by ioHub only: a snapshot of @p channels channels of @p frames
183 /// frames at @p sampleRate, for node @p owner saving buffer
184 /// @p bufferId to @p file.

186 std::size_t frames, double sampleRate);
187
188 /// The ID of the buffer saved.
189 const std::string& bufferId() const noexcept { return bufferId_; }
190 /// The file written, relative to the project root.
191 const std::string& file() const noexcept { return file_; }
192 /// The snapshot's sample rate, in hertz.
193 double sampleRate() const noexcept { return snapshot_.sampleRate(); }
194
195 /// Audio thread: starts a snapshot when no save is pending. False,
196 /// counting nothing, when one is.
197 bool begin() noexcept ESPRESSO_NONBLOCKING;
198 /// Audio thread: channel @p c of the snapshot being copied.
199 std::span<float> channel(int c) noexcept ESPRESSO_NONBLOCKING { return snapshot_.channel(c); }
200 /// Audio thread: the snapshot is complete, for the control thread to
201 /// write.
202 void finish() noexcept ESPRESSO_NONBLOCKING;
203
204 /// What writes a complete snapshot: the port, and the snapshot.
205 using writer = std::function<void(const savePort&, const sampleBuffer&)>;
206 /// Control thread: when a snapshot is complete, passes it to @p write,
207 /// then frees the port for the next save, also when @p write throws.
208 void writeReady(const writer& write);
209
210private:
211 enum class state : std::uint8_t { idle, copying, ready, writing };
212
213 std::string bufferId_;
214 std::string file_;
215 sampleBuffer snapshot_;
216 std::atomic<state> state_{state::idle};
217};
