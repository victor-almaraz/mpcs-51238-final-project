1/// @file
2/// `proj::jthread`: a thread that asks itself to stop and joins when it is
3/// destroyed, with the interface of C++20's `std::jthread` that the
4/// project uses. Unlike `std::jthread`, every thread gets a stack of
5/// `threadStackBytes`: secondary-thread stacks otherwise differ by
6/// platform (512 KiB on macOS; RLIMIT_STACK, or 2 MiB when unlimited, with
7/// glibc), and the interpreter's worker needs a measured size
8/// (`interpreter::maxEvalDepth`). Header-only. It lives in `protocol`, the
9/// lowest module that every component with a thread links.

47/// A thread that requests a stop and joins on destruction.
48class jthread {
49public:
50 /// No thread.
51 jthread() noexcept = default;
52
53 /// Runs @p f on a new thread with a stack of threadStackBytes. @p f may
54 /// take a stop_token as its only argument. Throws std::system_error if
55 /// the thread cannot be created.
56 template <typename F>
57 requires(!std::is_same_v<std::remove_cvref_t<F>, jthread>) // as std::jthread: never a copy
58 explicit jthread(F&& f) : flag_(std::make_shared<std::atomic<bool>>(false)) {
59 body_ = std::make_unique<std::function<void()>>(
60 [fn = std::forward<F>(f), token = stop_token(flag_)]() mutable {
61 // fn is a non-const copy and is called as an lvalue.
62 if constexpr (std::is_invocable_v<std::decay_t<F>&, stop_token&>) {
63 fn(token);
64 } else {
65 fn();
66 }
67 });
68 start();
69 }
70
71 /// Requests a stop and joins.
72 ~jthread() { stopAndJoin(); }
73
74 /// Takes over @p other's thread.
75 jthread(jthread&& other) noexcept
76 : flag_(std::move(other.flag_)),
77 body_(std::move(other.body_)),
78 thread_(other.thread_),
79 joinable_(std::exchange(other.joinable_, false)) {}
80 /// Stops and joins this thread, then takes over @p other's.
81 jthread& operator=(jthread&& other) noexcept {
82 if (this != &other) {
83 stopAndJoin();
84 flag_ = std::move(other.flag_);
85 body_ = std::move(other.body_);
86 thread_ = other.thread_;
87 joinable_ = std::exchange(other.joinable_, false);
88 }
89 return *this;
90 }
91 jthread(const jthread&) = delete;

94 /// Asks the thread to stop. Returns false when there is no thread or a
95 /// stop was already requested.
96 bool request_stop() noexcept { return flag_ && !flag_->exchange(true, std::memory_order_acq_rel); }
97
98 /// The token the thread function receives.
99 stop_token get_stop_token() const noexcept { return stop_token(flag_); }
100
101 /// True while the thread can be joined.
102 bool joinable() const noexcept { return joinable_; }
103 /// Waits for the thread to finish. Throws std::system_error if there is
104 /// no thread to join.
105 void join() {
106 if (!joinable_) throw std::system_error(std::make_error_code(std::errc::invalid_argument), "join");
107 const int result = ::pthread_join(thread_, nullptr); // no result wanted
108 joinable_ = false;
109 if (result != 0) throw std::system_error(result, std::generic_category(), "pthread_join");
110 }
111 /// The platform's thread handle.
112 pthread_t native_handle() const noexcept { return thread_; }
113
114private:
115 // POSIX threads pass the start routine's argument as void*: the narrow
116 // unsafe use at this boundary (modern-cpp skill, section 9). It is
117 // body_, which this object owns until the thread is joined.
118 static void* run(void* body) {
119 (*static_cast<std::function<void()>*>(body))();
120 return nullptr;
121 }
122
123 void start() {
124 pthread_attr_t attributes;
125 if (const int r = ::pthread_attr_init(&attributes); r != 0) {
126 throw std::system_error(r, std::generic_category(), "pthread_attr_init");
127 }
128 int result = ::pthread_attr_setstacksize(&attributes, threadStackBytes);
129 if (result == 0) result = ::pthread_create(&thread_, &attributes, &jthread::run, body_.get());
130 ::pthread_attr_destroy(&attributes);
131 if (result != 0) throw std::system_error(result, std::generic_category(), "pthread_create");
132 joinable_ = true;
133 }
134
135 void stopAndJoin() noexcept {
136 if (!joinable_) return;
137 request_stop();
138 ::pthread_join(thread_, nullptr);
139 joinable_ = false;
140 }
