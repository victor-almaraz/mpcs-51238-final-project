121/// A parameter value: a number in canonical units, a choice or text, a
122/// list of numbers, a bound pipe (`primitives.md` §7.1.2), a buffer
123/// (`samples.md` §12.2), or a list of strings (`io.md` §9.1).
124using paramValue =
125 std::variant<double, std::string, std::vector<double>, pipeRef, bufferRef, std::vector<std::string>>;

127/// The alternative @p v holds when it is a T: a checked view instead of
128/// `std::get_if`, which returns a pointer.
129template <typename T>
130optionalRef<T> as(const paramValue& v) noexcept {
131 if (!std::holds_alternative<T>(v)) return std::nullopt;
132 return std::cref(std::get<T>(v));
133}
134
135/// Deleted: the view would outlive the temporary it refers to.
136template <typename T>
137optionalRef<T> as(const paramValue&&) = delete;

166/// One element of a page's layout (`dsl-ui.md` §7.1): a group, which

227/// The pipe declarations of a graph, sorted by pipe: findPipe() for code
228/// that looks up many pipes of one graph. Building the index sorts
229/// references to the n declarations once; each lookup is then a binary
230/// search, about log2(n) comparisons, against findPipe()'s scan of up to
231/// n. The graph must outlive the index, and its pipes must not change.
232class pipeIndex {
233public:
234 /// Indexes the pipes @p graph declares.
235 explicit pipeIndex(const graphSpec& graph);
236 /// Deleted: the index would refer to a temporary graph.
237 explicit pipeIndex(const graphSpec&&) = delete;
238
239 /// The declaration of @p ref, if the graph has one: the first, as
240 /// findPipe() finds it, when the graph declares @p ref more than once.
241 optionalRef<pipeDecl> find(pipeRef ref) const noexcept;
242
243private:
244 /// By pipe; declarations of one pipe in the graph's order.
245 std::vector<std::reference_wrapper<const pipeDecl>> sorted_;
246};
