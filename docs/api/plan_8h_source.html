48/// An executable graph: primitive instances in execution order, the
49/// pipes they use, and their I/O ports. Shared between the engine's
50/// threads; its last owner is always on the control thread.
51class plan {
52public:
53 /// Plan ID assigned by the client.
54 std::int32_t id = 0;
55 /// Primitives, in execution order (§10.1). Shared with the transitions
56 /// that run them during a fade.
57 std::vector<std::shared_ptr<primitive>> nodes;
58 /// Per-node data, parallel to `nodes`.
59 std::vector<nodeInfo> info;
60 /// Video-domain primitives, in execution order: the nodes that read or
61 /// write a video pipe. The render thread runs them once per frame; the
62 /// audio thread never does (§3).
63 std::vector<std::shared_ptr<primitive>> videoNodes;
64 /// The frame the visual layer shows: the input of the first
65 /// `videoOut`, or null.
66 std::shared_ptr<const vPipe> display;
67 /// Pipe declarations applied when the plan becomes active.
68 std::vector<metadata::pipeDecl> pipes;
69 /// Every pipe the plan touches, sorted and unique.
70 std::vector<metadata::pipeRef> used;
71 /// The plan's feedback pipes, sorted.
72 std::vector<metadata::pipeRef> feedback;
73 /// I/O ports of the plan's sources and sinks, shared with them.
74 std::vector<std::shared_ptr<ioPort>> ports;
75 /// Tempo to set on activation.
76 std::optional<double> tempo;
77 /// Beats per bar.
78 int meter = 4;
79
80 /// True when pipe @p ref is a feedback pipe in this plan.
81 bool isFeedback(metadata::pipeRef ref) const noexcept;
82};
