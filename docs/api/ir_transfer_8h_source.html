34/// Reassembles IR announced by `/ir.begin`, one transfer per plan ID.
35/// Used by the server's network thread. At most `maxPendingTransfers`
36/// transfers, announcing at most `maxIrBytes` in total, are kept: a new
37/// announcement abandons the oldest transfers until it fits, so transfers
38/// that a client never finishes cannot accumulate.
39class irAssembler {
40public:
41 /// Outcome of finish().
42 struct result {
43 /// Outcome kinds.
44 enum class kind : std::uint8_t {
45 complete, ///< Every chunk arrived and the CRC-32 matches
46 missing, ///< Chunks are missing; see `missing`
47 checksumMismatch, ///< The CRC-32 differs from `/ir.begin`
48 incomplete, ///< The assembled size differs from `/ir.begin`
49 unknownPlan, ///< No `/ir.begin` for the plan
50 };
51 kind what; ///< The outcome
52 std::string payload; ///< For complete
53 std::vector<std::int32_t> missing; ///< For missing: chunk indices
54 };
55
56 /// Starts plan @p planId. Replaces an earlier transfer of the same
57 /// plan and abandons the oldest transfers beyond the limits. Returns
58 /// false when the announcement is invalid.
59 bool begin(std::int32_t planId, std::int64_t totalBytes, std::int32_t chunkCount, std::uint32_t crc);
60
61 /// Stores chunk @p index of @p planId. Returns false for an unknown
62 /// plan, an index out of range, or an oversized chunk.
63 bool add(std::int32_t planId, std::int32_t index, std::span<const std::uint8_t> data);
64
65 /// Checks that every chunk arrived and the CRC-32 matches. A complete
66 /// or failed transfer is forgotten; a transfer with missing chunks is
67 /// kept so the client can resend them (`protocol.md` §6.3).
68 result finish(std::int32_t planId);
69
70 /// Number of transfers in progress.
71 std::size_t pending() const noexcept { return transfers_.size(); }
72
73 /// Transfers abandoned to stay within the limits.
74 std::size_t abandoned() const noexcept { return abandoned_; }
75
76private:

80 std::uint64_t started = 0; // order of /ir.begin, oldest first
81 std::vector<std::optional<std::vector<std::uint8_t>>> chunks;
82 };
83 std::map<std::int32_t, transfer> transfers_;
84 std::uint64_t nextStart_ = 0;
