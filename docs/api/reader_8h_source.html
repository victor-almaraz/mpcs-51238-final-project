26/// Kind of a syntax-tree node.

31/// A unit literal as written: its numeric value and suffix. Conversion

40/// A syntax-tree node. `'x` reads as the list `(quote x)`.
41struct datum {
42 /// Kind of node.
43 datumKind kind = datumKind::nil;
44 /// Location in the source.
45 sourceSpan span;
46 /// Value of a boolean.
47 bool boolean = false;
48 /// Value of an integer.
49 std::int64_t integer = 0;
50 /// Value of a float.
51 double floating = 0.0;
52 /// Value of a unit literal.
53 quantityLiteral quantity;
54 std::string text; ///< String contents (unescaped), or keyword/symbol name
55 std::vector<datum> items; ///< List or vector elements
56
57 /// True when this datum is a symbol named @p name.
58 bool isSymbol(std::string_view name) const noexcept {
59 return kind == datumKind::symbol && text == name;
60 }
61};
62
63/// A read error with the span of the offending text.
64struct readError {
65 /// Description of the error.
66 std::string message;
67 /// Location of the offending text.
68 sourceSpan span;
69};
70
71/// Result of reading a source text.
72struct readResult {
73 std::vector<datum> data; ///< Top-level data, in order
74 std::optional<readError> error; ///< Set when reading failed
75};
76
77/// Reads every datum in @p source using the strict lexer.
