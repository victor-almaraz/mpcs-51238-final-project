24/// Decoded audio: what a sample loader returns (`samples.md` §3.3), at
25/// the rate it was asked for or, asked for none, at the file's own.
26struct decodedAudio {
27 /// One vector of samples per channel, all of one length.
28 std::vector<std::vector<float>> channels;
29 /// The rate of the samples, in hertz.
30 double sampleRate = 0.0;
31
32 /// The length, in frames: that of the first channel, or 0.
33 std::size_t frames() const noexcept { return channels.empty() ? 0 : channels.front().size(); }
34};

80/// Channel @p to of @p toChannels, taken from a source of @p fromChannels
81/// channels whose channel c is @p get(c) (`samples.md` §4.4): with equal
82/// counts, channel @p to; from a mono source, its one channel; into a mono
83/// destination, the average; otherwise the same channel, or silence past
84/// the source's last.
85template <typename Get>
86float mapChannel(int to, int toChannels, int fromChannels, Get get) noexcept ESPRESSO_NONBLOCKING {
87 if (fromChannels == toChannels || fromChannels == 1) return get(fromChannels == 1 ? 0 : to);
88 if (toChannels == 1) {
89 float sum = 0.0f;
90 for (int c = 0; c < fromChannels; ++c) sum += get(c);
91 return sum / static_cast<float>(fromChannels);
92 }
93 return to < fromChannels ? get(to) : 0.0f;
94}
