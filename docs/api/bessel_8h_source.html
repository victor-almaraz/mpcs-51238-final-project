19/// I0(@p x), the modified Bessel function of the first kind of order 0, by
20/// its power series: the sum of (x / 2)^2k / (k!)^2 from k = 0, adding
21/// terms while the last one added exceeds @p tolerance times the sum, and
22/// at most 63 terms past the first. A Kaiser window of shape β is
23/// I0(β √(1 − r²)) / I0(β) at r from −1 to 1.
24constexpr double besselI0(double x, double tolerance) noexcept {
25 double sum = 1.0;
26 double term = 1.0;
27 for (int k = 1; k < 64 && term > tolerance * sum; ++k) {
28 const double half = x / (2.0 * k);
29 term *= half * half;
30 sum += term;
31 }
32 return sum;
33}

35/// A Kaiser-windowed sinc of @p Zeros zero crossings either side of its
36/// centre, as a table of @p Resolution points a crossing of @p T values,
37/// read by linear interpolation between points. Point i holds
38/// sinc(u) × I0(β √(1 − (u / Zeros)²)) / I0(β) at u = i / Resolution
39/// crossings, computed in double, and a 0 follows the last point. Made
40/// where allocation is allowed; read on any thread.
41template <typename T, int Zeros, int Resolution>
42class kaiserSinc {
43public:
44 /// The last point: the kernel's reach, in points from its centre.
45 static constexpr int reach = Zeros * Resolution;
46
47 /// The kernel of window shape @p beta, whose series (besselI0) stops
48 /// at terms @p tolerance times its sum.
49 kaiserSinc(double beta, double tolerance) noexcept {
50 const double norm = besselI0(beta, tolerance);
51 for (int i = 0; i <= reach; ++i) {
52 const double u = static_cast<double>(i) / Resolution;
53 const double r = u / Zeros;
54 const double sinc = u == 0.0 ? 1.0 : std::sin(std::numbers::pi * u) / (std::numbers::pi * u);
55 table_[static_cast<std::size_t>(i)] =
56 static_cast<T>(sinc * besselI0(beta * std::sqrt(1.0 - (r * r)), tolerance) / norm);
57 }
58 }
59
60 /// The kernel at @p p points from its centre, from 0 to `reach` + 1.
61 T at(T p) const noexcept ESPRESSO_NONBLOCKING {
62 const int i = std::min(static_cast<int>(p), reach); // p >= 0; int converts in one instruction
63 const T frac = p - static_cast<T>(i);
64 const auto k = static_cast<std::size_t>(i);
65 return table_[k] + (frac * (table_[k + 1] - table_[k]));
66 }
67
68private:
69 std::array<T, static_cast<std::size_t>(reach) + 2> table_{};
70};
