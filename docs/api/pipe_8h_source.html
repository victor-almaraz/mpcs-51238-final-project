30/// Pipe interface (`primitives.md` §5.1). Primitives fetch a pipe's data
31/// once per block and never call virtual methods per sample.
32class pipe {
33public:
34 pipe() = default;
35 virtual ~pipe() = default;
36 // Pipes are shared through the pipe table, never copied or moved.
37 pipe(const pipe&) = delete;
38 pipe& operator=(const pipe&) = delete;
39 pipe(pipe&&) = delete;
40 pipe& operator=(pipe&&) = delete;
41
42 /// The pipe's type.
43 virtual metadata::pipeKind kind() const noexcept ESPRESSO_NONBLOCKING = 0;
44
45 /// Active channel count, from 1 to `metadata::maxChannels`.
46 int channels() const noexcept { return channels_; }
47 /// True when the pipe delays its data by one block (§5.6).
48 bool feedback() const noexcept { return feedback_; }
49
50 /// Sets the channel count, clamped to 1 to `metadata::maxChannels`, and
51 /// the feedback flag. Called on the audio thread when a plan becomes
52 /// active; storage is preallocated.
53 virtual void configure(int channels, bool feedback) noexcept ESPRESSO_NONBLOCKING;
54
55protected:
56 /// Active channel count.
57 int channels_ = 1;
58 /// Feedback flag.
59 bool feedback_ = false;
60};

62/// Audio pipe: `channels` × block samples. Writers sum; the engine zeroes
63/// the pipe at each block start (§5.2).
64class aPipe final : public pipe {
65public:
66 /// Preallocates `maxChannels` channels of @p blockSize samples, twice
67 /// (the second buffer serves feedback).
68 explicit aPipe(int blockSize);
69
70 /// Audio.
71 metadata::pipeKind kind() const noexcept ESPRESSO_NONBLOCKING override {
72 return metadata::pipeKind::audio;
73 }
74 /// Sets the channel count and feedback flag. A change of either clears
75 /// both buffers, so no channel or feedback buffer replays old samples.
76 void configure(int channels, bool feedback) noexcept ESPRESSO_NONBLOCKING override;
77
78 /// Samples of @p channel in the current block; empty for a channel
79 /// outside 0 to `maxChannels` − 1. A feedback pipe returns the previous
80 /// block's data.
81 std::span<const float> read(int channel) const noexcept ESPRESSO_NONBLOCKING;
82
83 /// Adds @p src times @p gain into @p channel. A channel outside the
84 /// active count is ignored.
85 void add(int channel, std::span<const float> src, float gain = 1.0f) noexcept ESPRESSO_NONBLOCKING;
86 /// Adds @p src into @p channel with a gain ramp from @p gainStart to
87 /// @p gainEnd across the block (crossfades, §9.3). A channel outside the
88 /// active count is ignored.
89 void addRamp(int channel, std::span<const float> src, float gainStart,
90 float gainEnd) noexcept ESPRESSO_NONBLOCKING;
91
92 /// Starts a block of @p numSamples: zeroes the buffer being written.
93 void beginBlock(int numSamples) noexcept ESPRESSO_NONBLOCKING;
94 /// Ends a block: publishes each channel's last sample (latest()), and
95 /// a feedback pipe swaps its buffers.
96 void endBlock() noexcept ESPRESSO_NONBLOCKING;
97
98 /// Any thread: the last sample of @p channel at the end of the latest
99 /// block. The render thread reads audio through it (the window of one
100 /// sample of `primitives.md` §8.1.2's latestWindow mode).
101 float latest(int channel) const noexcept {
102 return latest_[static_cast<std::size_t>(std::clamp(channel, 0, metadata::maxChannels - 1))].load(
103 std::memory_order_relaxed);
104 }
105
106private:
107 // A feedback pipe reads one buffer while writers fill the other and
108 // swaps them at each block end; any other pipe uses the first only.
109 const std::vector<float>& readBuffer() const noexcept { return feedback_ && swapped_ ? second_ : first_; }
110 std::vector<float>& writeBuffer() noexcept { return !feedback_ ? first_ : (swapped_ ? first_ : second_); }
111
112 // The samples of @p channel in @p buffer (const or not).
113 template <typename Buffer>
114 auto channelOf(Buffer& buffer, int channel) const noexcept {
115 const auto size = static_cast<std::size_t>(blockSize_);
116 return std::span(buffer).subspan(static_cast<std::size_t>(channel) * size, size);
117 }
118 // The part of @p channel this block writes, or empty when the channel
119 // is not active.
120 std::span<float> written(int channel) noexcept ESPRESSO_NONBLOCKING;
121
122 int blockSize_ = 0;
123 int numSamples_ = 0;
124 std::vector<float> first_;
125 std::vector<float> second_;
126 bool swapped_ = false;
127 std::array<std::atomic<float>, metadata::maxChannels> latest_{};
128};

130/// Control pipe: one value per channel. A write replaces the value, which
131/// holds until the next write (§5.3). Values are atomics so other threads
132/// can read them (`/c.get`).
133class cPipe final : public pipe {
134public:
135 /// Control.
136 metadata::pipeKind kind() const noexcept ESPRESSO_NONBLOCKING override {
137 return metadata::pipeKind::control;
138 }
139 /// Sets the channel count and feedback flag. Channels that become
140 /// active take channel 0's value, which a reader of the narrower pipe
141 /// saw on every channel; a feedback change carries the current values
142 /// into both halves, so the next block end publishes no stale value.
143 void configure(int channels, bool feedback) noexcept ESPRESSO_NONBLOCKING override;
144
145 /// The value of @p channel. A feedback pipe returns the value written
146 /// in the previous block.
147 float value(int channel = 0) const noexcept {
148 return front_[clampChannel(channel)].load(std::memory_order_relaxed);
149 }
150
151 /// Sets @p channel to @p v.
152 void set(int channel, float v) noexcept ESPRESSO_NONBLOCKING;
153 /// Sets every channel to @p v.
154 void setAll(float v) noexcept ESPRESSO_NONBLOCKING;
155
156 /// Ends a block: a feedback pipe publishes the values written in it.
157 void endBlock() noexcept ESPRESSO_NONBLOCKING;
158
159private:
160 static std::size_t clampChannel(int c) noexcept {
161 return static_cast<std::size_t>(c < 0 ? 0
162 : (c >= metadata::maxChannels ? metadata::maxChannels - 1 : c));
163 }
164
165 // Value-initialized: every value starts at 0.
166 std::array<std::atomic<float>, metadata::maxChannels> front_{};
167 std::array<std::atomic<float>, metadata::maxChannels> back_{};
168};

170/// Event pipe: a bounded array of events sorted by sample offset, plus a

178 /// Overflows increment @p overflowCounter, which the pipe shares.

180
181 /// Event.

183 return metadata::pipeKind::event;
184 }
185
186 /// This block's events, sorted by sample offset. Equal offsets keep
187 /// writer order (§5.4.2).

189
190 /// Inserts @p e in offset order. When full, drops it, counts an
191 /// overflow, and returns false (§5.4.3).
192 bool write(const event& e) noexcept ESPRESSO_NONBLOCKING;
193
194 /// Copies an OSC message into the pool and writes an OSC event for
195 /// it. Drops and counts it when the pool or the array is full.

197
198 /// The pool bytes of @p ref.
199 std::span<const std::byte> data(poolRef ref) const noexcept ESPRESSO_NONBLOCKING;
200
201 /// Copies @p e, and its pool data for an OSC event, from @p from.
202 bool copyFrom(const ePipe& from, const event& e) noexcept ESPRESSO_NONBLOCKING;
203
204 /// Starts a block: clears the events and the pool.
205 void beginBlock() noexcept ESPRESSO_NONBLOCKING;
206
207private:
208 std::optional<poolRef> store(std::span<const std::byte> bytes) noexcept ESPRESSO_NONBLOCKING;
209 void overflow() noexcept ESPRESSO_NONBLOCKING;
210
211 std::vector<event> events_;
212 std::size_t count_ = 0;
213 std::vector<std::byte> pool_;
214 std::size_t poolUsed_ = 0;
215 counter overflows_;
216};
