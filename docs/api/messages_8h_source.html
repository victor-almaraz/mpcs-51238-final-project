113/// Error codes carried by `/fail` (`protocol.md` §8).
114enum class errorCode : std::int32_t {

128/// True when replies to @p addr carry the request ID as their first
129/// argument.
130constexpr bool isReply(std::string_view addr) noexcept {
131 return addr == address::done || addr == address::fail || addr == address::helloOk ||
132 addr == address::statusReply || addr == address::irMissing || addr == address::controlValue ||
133 addr == address::synced;
134}

136/// What a `/fail` names in its trailing arguments (`protocol.md` §5).
137enum class failSubject : std::uint8_t {
138 none, ///< The plan or request as a whole
139 node, ///< One node
140 buffer, ///< One buffer
141};
142
143/// The name `/fail` carries for @p s: `node` or `buffer`; empty for none.
144constexpr std::string_view subjectName(failSubject s) noexcept {
145 switch (s) {
146 case failSubject::node: return "node";
147 case failSubject::buffer: return "buffer";
148 case failSubject::none: break;
149 }
150 return {};
151}
152
153/// The subject that @p name names; none for any other text.
154constexpr failSubject subjectNamed(std::string_view name) noexcept {
155 if (name == subjectName(failSubject::node)) return failSubject::node;
156 if (name == subjectName(failSubject::buffer)) return failSubject::buffer;
157 return failSubject::none;
158}

160/// Why a plan was refused (`protocol.md` §5): the message, and, when one
161/// node or buffer of the plan failed, which, so a client can find its
162/// source.
163struct planError {
164 /// What is wrong, as `/fail` reports it.
165 std::string message;
166 /// What failed; none for the plan as a whole.
167 failSubject subject = failSubject::none;
168 /// The node's or buffer's ID; empty for the plan as a whole.
169 std::string id;
170
171 /// An error about the plan as a whole.
172 static planError plain(std::string message) {
173 return {.message = std::move(message), .subject = failSubject::none, .id = {}};
174 }
175};
