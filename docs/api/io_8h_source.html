75/// A queue between one primitive and the I/O layer. Shared by the
76/// primitive and its plan; the hub keeps only a weak reference, so a port
77/// is gone when its plan is. Input ports accept data only while active,
78/// so a plan that is loaded but not yet playing does not accumulate stale
79/// messages. An input port names the node that reads it: the hub delivers
80/// each message to one active port per node, so an adopting node and the
81/// node it replaces, both active during a swap, never receive it twice.
82class ioPort {
83public:
84 virtual ~ioPort() = default;
85
86 ioPort(const ioPort&) = delete;
87 ioPort& operator=(const ioPort&) = delete;
88
89 /// What the port connects to.
90 portKind kind() const noexcept { return kind_; }
91 /// Node ID of the source reading an input port; empty for outputs.
92 const std::string& owner() const noexcept { return owner_; }
93 /// MIDI device name pattern; empty matches every device.
94 const std::string& device() const noexcept { return device_; }
95 /// UDP port, for OSC ports.
96 std::uint16_t udpPort() const noexcept { return udpPort_; }
97
98 /// Activates or deactivates the port (audio thread, at a swap).
99 void setActive(bool on) noexcept ESPRESSO_NONBLOCKING { active_.store(on, std::memory_order_seq_cst); }
100 /// True while the port's plan is running.
101 bool active() const noexcept { return active_.load(std::memory_order_seq_cst); }
102
103 /// A delivery in progress (hub thread), from before it reads whether
104 /// the port is active until after it pushes. The audio thread may
105 /// deactivate the port in between; closed() tells it when no delivery
106 /// can still push.
107 class delivery {
108 public:
109 /// Starts a delivery to @p port.
110 explicit delivery(ioPort& port) noexcept : port_(port) {
111 port_.delivering_.fetch_add(1, std::memory_order_seq_cst);
112 }
113 ~delivery() { port_.delivering_.fetch_sub(1, std::memory_order_seq_cst); }
114
115 delivery(const delivery&) = delete;
116 delivery& operator=(const delivery&) = delete;
117 delivery(delivery&&) = delete;
118 delivery& operator=(delivery&&) = delete;
119
120 /// True when the port was active once the delivery had started.
121 bool admitted() const noexcept { return port_.active(); }
122
123 private:
124 ioPort& port_;
125 };
126
127 /// True when no delivery can push into the port any more: it is
128 /// inactive and none is in progress. A delivery that starts later
129 /// reads the port inactive and pushes nothing (sequentially consistent
130 /// order).
131 bool closed() const noexcept ESPRESSO_NONBLOCKING {
132 return !active() && delivering_.load(std::memory_order_seq_cst) == 0;
133 }
134
135protected:
136 /// A port of @p kind for @p device or @p udpPort, read by node @p owner.
137 ioPort(portKind kind, std::string owner, std::string device, std::uint16_t udpPort)
138 : kind_(kind), owner_(std::move(owner)), device_(std::move(device)), udpPort_(udpPort) {}
139
140private:
141 portKind kind_ = portKind::midiIn;
142 std::string owner_;
143 std::string device_;
144 std::uint16_t udpPort_ = 0;
145 std::atomic<bool> active_{false};
146 std::atomic<int> delivering_{0};
147};

173/// Registry of ports. Opening ports happens on the builder thread;
174/// delivery and draining happen on I/O threads. They lock the hub to find
175/// ports, never while calling back; the audio thread never locks it.
176class ioHub {
177public:
178 /// Opens a MIDI input for devices matching @p device, read by node
179 /// @p owner.
180 std::shared_ptr<midiPort> openMidiIn(std::string owner, std::string device);
181 /// Opens a MIDI output for the device matching @p device.
182 std::shared_ptr<midiPort> openMidiOut(std::string device);
183 /// Opens an OSC input on UDP port @p port, read by node @p owner.
184 std::shared_ptr<oscPort> openOscIn(std::string owner, std::uint16_t port);
185 /// Opens an OSC output to UDP port @p port.
186 std::shared_ptr<oscPort> openOscOut(std::uint16_t port);
187
188 /// I/O thread: queues @p packet for each node with an active MIDI
189 /// input matching @p deviceName, in that node's oldest active port.
190 /// Full queues drop it and count a drop. Deliveries hold the hub lock,
191 /// so each queue has one producer at a time.
192 void deliverMidi(std::string_view deviceName, const midiPacket& packet);
193 /// I/O thread: queues @p packet for each node with an active OSC input
194 /// on @p port, as deliverMidi() does.
195 void deliverOsc(std::uint16_t port, const oscPacket& packet);
196
197 /// I/O thread: passes every queued MIDI output packet to @p send with
198 /// the port's device pattern. Called from one thread at a time; @p send
199 /// runs without the hub lock and may call the hub.
200 void drainMidiOut(const std::function<void(std::string_view device, const midiPacket&)>& send);
201 /// I/O thread: passes every queued OSC output packet to @p send with
202 /// its destination port. Called from one thread at a time; @p send runs
203 /// without the hub lock and may call the hub.
204 void drainOscOut(const std::function<void(std::uint16_t port, const oscPacket&)>& send);
205
206 /// UDP ports that OSC inputs listen on, sorted and unique.
207 std::vector<std::uint16_t> oscInputPorts() const;
208
209 /// Audio thread: reports a fault in node @p node. Real-time safe; a
210 /// full queue drops the report but still counts it.
211 void reportFault(std::string_view node, std::string_view what) noexcept ESPRESSO_NONBLOCKING;
212 /// Control thread: removes the reported faults, each as
213 /// `node ID: what`.
214 std::vector<std::string> takeFaults();

218 /// Counts a dropped packet (full queue). Real-time safe.
219 void countDrop() noexcept ESPRESSO_NONBLOCKING { drops_.fetch_add(1, std::memory_order_relaxed); }
220 /// Packets dropped by full queues.
221 std::uint64_t drops() const noexcept { return drops_.load(std::memory_order_relaxed); }

228 // Calls @p visit with each live port of @p kind, as its concrete type,
229 // oldest first. The caller holds the hub lock. A port whose plan is
230 // gone is dropped.
231 template <typename Port, typename Visit>
232 void forEachLocked(portKind kind, Visit&& visit) const;
233 // The live ports of @p kind, found under the hub lock.
234 template <typename Port>
235 std::vector<std::shared_ptr<Port>> live(portKind kind) const;
236 // Pushes @p packet into the oldest active port of each owner among the

241 template <typename Port, typename Packet, typename Send>
