25/// What a project file declares. Each member not given keeps its default.
26struct projectSpec {
27 /// `:name`: the project's name.
28 std::string name;
29 /// `:entry`: the piece to build.
30 std::string entry = "main";
31 /// `:seed`: the seed for random generators (`interpreter.md` §13.7).
32 std::uint64_t seed = 0;
33 /// `:page-port`: the TCP port of the project's browser front-end
34 /// (`io.md` §12.2); none opens no page listener.
35 std::optional<int> pagePort;
36};

46/// Reads @p source, the text of a project file: one form,
47/// `(project :name "drift" :root "." :entry main :seed 7 :page-port 8000)`,
48/// whose arguments are all optional. `:root` may only be `"."`, the
49/// folder that holds the file. Never throws on bad input.

60/// The options for compiling a piece of a project: @p options with the
